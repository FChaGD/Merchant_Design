```csharp
namespace Game.Core
{
    /// <summary>
    /// 게임 날짜(Docs/기획/시스템/84번 §3). 누적 일수 하나만 상태로 갖고 월·일은 계산한다 - 저장/불러오기가 생겨도 정수 하나만 저장하면 된다.
    /// 1개월 30일·1년 6개월은 기획 고정 규칙이라 상수다. 절대 연도는 없고 경과 해 수는 내부에서만 쓴다(표시하지 않음).
    /// </summary>
    public readonly struct GameDate
    {
        public const int DaysPerMonth = 30;
        public const int MonthsPerYear = 6;
        public const int DaysPerYear = DaysPerMonth * MonthsPerYear;

        public int TotalDays { get; }

        public GameDate(int totalDays)
        {
            TotalDays = totalDays < 0 ? 0 : totalDays;
        }

        public int ElapsedYears => TotalDays / DaysPerYear;
        public int Month => TotalDays % DaysPerYear / DaysPerMonth + 1;
        public int Day => TotalDays % DaysPerMonth + 1;

        public string ToDisplayString() => $"{Month}월 {Day}일";
    }
}
```

```csharp
using System;

namespace Game.Core
{
    /// <summary>
    /// 날짜 보관·진행(Docs/설계/85번 §3.1). DateManager가 감싸는 순수 객체 - Unity 없이 테스트하기 위해 분리했다.
    /// 실제로 바뀐 경우에만 알린다(0 이하 진행은 무시).
    /// </summary>
    public sealed class GameCalendar
    {
        public GameDate Today { get; private set; }

        public event Action Changed;

        public GameCalendar(int totalDays = 0)
        {
            Today = new GameDate(totalDays);
        }

        public void AdvanceDays(int days)
        {
            if (days <= 0) return;

            Today = new GameDate(Today.TotalDays + days);
            Changed?.Invoke();
        }
    }
}
```

```csharp
using UnityEngine;

namespace Game.Core
{
    /// <summary>
    /// 상행 이동 초 → 게임 일수(Docs/기획/시스템/84번 §4.1, 설계 85번 §3.2). 정보 패널 예상과 실제 가산이 같은 이동 수치를 읽어야 일치하므로
    /// 환산 값은 날짜 매니저가 아니라 TripTravelSettings에 둔다. 설정값이 0 이하면(새 필드가 없는 옛 씬 저장값) 기본값을 쓴다.
    /// 구간 소요시간은 공식 계산값(float)이라 89.99999초처럼 경계 바로 아래로 나올 수 있어, 버리기 전에 아주 작은 보정을 더한다.
    /// </summary>
    public static class TravelDayRules
    {
        private const float FloorEpsilonDays = 1e-4f;

        public static int ToDays(TripTravelSettings settings, float seconds) => FloorDays(seconds / SecondsPerDay(settings));

        public static int ToRoutDays(TripTravelSettings settings, float seconds) => FloorDays(seconds * RoutMultiplier(settings) / SecondsPerDay(settings));

        private static float SecondsPerDay(TripTravelSettings settings)
            => settings != null && settings.SecondsPerDay > 0f ? settings.SecondsPerDay : TripTravelSettings.DefaultSecondsPerDay;

        private static float RoutMultiplier(TripTravelSettings settings)
            => settings != null && settings.RoutDayMultiplier > 0f ? settings.RoutDayMultiplier : TripTravelSettings.DefaultRoutDayMultiplier;

        private static int FloorDays(float days) => days <= 0f ? 0 : Mathf.FloorToInt(days + FloorEpsilonDays);
    }
}
```

```csharp
namespace Game.Core
{
    /// <summary>
    /// 게임 날짜 전역 매니저(Docs/설계/85번 §3.1). Bootstrap 상주, 조회·진행 계약을 각 타입으로 따로 등록한다(상위·하위 타입 등록으로는
    /// 조회되지 않음). 상행 규칙(초→일)은 모른다 - 누가 며칠을 더할지는 소비자(TripDateAccrual 등)가 정한다.
    /// 저장/불러오기가 없어 ResolveDependencies마다 1월 1일로 시작한다. 달력을 새로 만들어도 소비자는 이 컴포넌트의 Changed를
    /// 구독하므로 구독이 끊기지 않는다(GoldHoldingsService와 같은 방식).
    /// </summary>
    public class DateManager : MonoBehaviour, IDateReader, IDateAdvancer, IManagedComponent
    {
        private GameCalendar calendar = new();

        public event Action Changed;

        public GameDate Today => calendar.Today;

        public void RegisterSelf(IDependencyRegistrar registrar)
        {
            registrar.Register<IDateReader>(this);
            registrar.Register<IDateAdvancer>(this);
        }

        public void ResolveDependencies(IDependencyResolver registrar)
        {
            calendar.Changed -= RaiseChanged;
            calendar = new GameCalendar();
            calendar.Changed += RaiseChanged;
        }

        public void AdvanceDays(int days) => calendar.AdvanceDays(days);

        private void RaiseChanged() => Changed?.Invoke();
    }
}
```
