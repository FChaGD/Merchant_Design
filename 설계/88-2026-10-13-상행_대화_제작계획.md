> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 상행 중 이동을 멈추지 않는 대화(20초 검사·상황 대화·랜덤 30%·수동 걸기)를 말풍선 + 좌우 스탠딩 화면으로 띄우고, 중단·일시정지·기록 규칙과 예시 콘텐츠 D1~D4, 디버그 도구까지 만든다.

**Architecture:** 판정·진행 규칙은 Unity를 모르는 순수 C#(`TripDialogueSelector`·`TripDialogueDirector`·`DialogueStageLayout`·`StagedDialogueView`)에 두고 EditMode로 검증한다. Bootstrap 쪽은 FieldManager 하위 MonoBehaviour 껍데기(`TripDialogueCoordinator`)가 DI·신호 구독·프레임 진행만 맡고, Field 씬의 화면(`FieldDialogueView`·`FieldManualTalkView`)은 받은 값을 그리기만 한다. 화면은 Field 로드마다 `FieldUIWiring`이 진행자에 다시 연결한다.

**Tech Stack:** Unity 6000.3, C#, NUnit(EditMode), TextMeshPro, Yarn Spinner 3.2.8, ClosedXML(에디터), Unity CLI(`ut.sh`, `unity command ...`)

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;

namespace Game.Core
{
    public enum DialogueSide { Left, Right }

    /// <summary>스탠딩 한 자리의 표시 상태. Visible=false면 그 자리를 숨긴다(대화에 나오지 않는 자리, 설계 88 §5.2).</summary>
    public readonly struct StandingState
    {
        public bool Visible { get; }
        public string Name { get; }
        public Color Color { get; }
        public bool Highlighted { get; }

        public StandingState(bool visible, string name, Color color, bool highlighted)
        {
            Visible = visible;
            Name = name ?? string.Empty;
            Color = color;
            Highlighted = highlighted;
        }

        public static StandingState Hidden => new(false, string.Empty, Color.clear, false);
    }

    /// <summary>
    /// 말풍선·좌우 스탠딩 화면 계약(설계 88 §5, 89 이벤트 대화 화면도 같은 계약). 자리·이름·색·밝기 결정은 StagedDialogueView가 하고,
    /// 구현은 받은 값을 그리기만 한다 - 자리 규칙을 Unity 없이 검증하기 위해서다.
    /// </summary>
    public interface IDialogueStageView
    {
        void SetVisible(bool visible);
        /// <summary>새 줄. 보이는 글자 수는 0부터 다시 시작한다.</summary>
        void SetBubble(string speakerName, string text);
        void SetVisibleCharacters(int count);
        void SetContinueMarker(bool shown);
        void SetStanding(DialogueSide side, StandingState state);
        void ShowOptions(IReadOnlyList<DialogueOptionData> options, Action<int> onChosen);
        void SetOptionTimeRemaining(float fraction01);
        void HideOptions();
    }

    /// <summary>
    /// 화자 자리(설계 88 §5.3) - 화자마다 첫 대사의 `#left`/`#right`로 자리가 정해지고 대화 끝까지 유지된다. 태그가 없거나 같은 자리를 두 화자가 쓰는 대본은
    /// 콘텐츠 검증이 막지만, 실행 중엔 멈추지 않도록 빈 자리(왼쪽 먼저)에 두거나 최근 화자로 덮어쓴다.
    /// </summary>
    public sealed class DialogueStageLayout
    {
        public string LeftSpeaker { get; private set; }
        public string RightSpeaker { get; private set; }

        /// <summary>말하는 자리. 서술 줄(화자 없음)은 null.</summary>
        public DialogueSide? Apply(DialogueLineData line)
        {
            var id = line.SpeakerId;
            if (string.IsNullOrEmpty(id)) return null;
            if (LeftSpeaker == id) return DialogueSide.Left;
            if (RightSpeaker == id) return DialogueSide.Right;

            DialogueSide side;
            if (line.HasTag(DialogueTags.Left)) side = DialogueSide.Left;
            else if (line.HasTag(DialogueTags.Right)) side = DialogueSide.Right;
            else side = LeftSpeaker == null ? DialogueSide.Left : DialogueSide.Right;

            if (side == DialogueSide.Left) LeftSpeaker = id;
            else RightSpeaker = id;
            return side;
        }

        public void Reset()
        {
            LeftSpeaker = null;
            RightSpeaker = null;
        }
    }
}
```

`SpeakerPresentation.cs`

```csharp
using System;
using UnityEngine;

namespace Game.Core
{
    /// <summary>
    /// 화자 표시(설계 88 §5.2). 이름은 캐릭터 테이블·외부 인물 순으로 찾고(조회 함수는 소유자가 넘긴다), 못 찾으면 Id를 그대로 보여 준다 - 빈 이름 칸보다
    /// 대본 오류를 드러내기 쉽다. 색은 Placeholder 스탠딩용으로 Id에서 결정한다(같은 인물 = 같은 색).
    /// </summary>
    public sealed class SpeakerPresentation
    {
        private const float Saturation = 0.45f;
        private const float Brightness = 0.85f;

        private readonly Func<string, string> nameOf;

        public SpeakerPresentation(Func<string, string> nameOf) => this.nameOf = nameOf;

        public string NameOf(string speakerId)
        {
            if (string.IsNullOrEmpty(speakerId)) return string.Empty;
            var name = nameOf?.Invoke(speakerId);
            return string.IsNullOrEmpty(name) ? speakerId : name;
        }

        // string.GetHashCode는 실행마다 달라질 수 있어 FNV-1a로 직접 계산한다 - 같은 인물이 상행마다 다른 색이 되지 않게.
        public static Color ColorOf(string speakerId)
        {
            unchecked
            {
                uint hash = 2166136261;
                foreach (var c in speakerId ?? string.Empty)
                {
                    hash ^= c;
                    hash *= 16777619;
                }
                return Color.HSVToRGB(hash % 360 / 360f, Saturation, Brightness);
            }
        }
    }
}
```

```csharp
        private static void ValidateSides(YarnScriptNode node, List<string> errors)
        {
            var sideOf = new Dictionary<string, string>();
            foreach (var line in node.Lines)
            {
                if (line.IsOption || line.IsCommand || line.SpeakerId.Length == 0 || sideOf.ContainsKey(line.SpeakerId)) continue;

                var side = line.Tags.Contains(DialogueTags.Left) ? DialogueTags.Left
                    : line.Tags.Contains(DialogueTags.Right) ? DialogueTags.Right
                    : null;
                if (side == null)
                {
                    errors.Add($"{node.Title}:{line.LineNumber} 화자 '{line.SpeakerId}'의 첫 대사에 #left 또는 #right가 없다.");
                    sideOf[line.SpeakerId] = string.Empty;
                    continue;
                }

                foreach (var pair in sideOf)
                {
                    if (pair.Value != side) continue;
                    errors.Add($"{node.Title}:{line.LineNumber} 화자 '{line.SpeakerId}'가 '{pair.Key}'와 같은 자리(#{side})를 쓴다.");
                    break;
                }
                sideOf[line.SpeakerId] = side;
            }
```

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    /// <summary>
    /// 상행 대화 고르기 규칙(설계 88 §3·§4.1). 20초 검사: 쌓인 일 중 '이벤트 직후'를 먼저, 그다음 먼저 생긴 순서로 상황 대화(테이블 순서 첫 행)를 찾고,
    /// 없으면 30% 판정 후 랜덤 대화 가중치 추첨. 수동 걸기: 그 상단원이 화자인 안 한 수동 대화(테이블 순서 첫 행) → 없으면 반복 대화 가중치 추첨.
    /// 횟수 제한·화자 고용·조건은 설계 87의 후보 거르기를 그대로 쓴다.
    /// </summary>
    public static class TripDialogueSelector
    {
        public const float RandomDialogueChance = 0.3f;

        public static bool TryMapMilestone(TripMilestone milestone, out DialogueTrigger trigger)
        {
            switch (milestone)
            {
                case TripMilestone.TripStarted: trigger = DialogueTrigger.Departure; return true;
                case TripMilestone.BattleEnded: trigger = DialogueTrigger.AfterBattle; return true;
                case TripMilestone.EventEnded: trigger = DialogueTrigger.AfterEvent; return true;
                case TripMilestone.MidCityArrived: trigger = DialogueTrigger.MidCityArrival; return true;
                default: trigger = DialogueTrigger.None; return false;
            }
        }

        public static DialogueEntry PickForCheck(IReadOnlyList<DialogueTrigger> pending, IReadOnlyList<DialogueEntry> dialogues, IEncounterHistoryReader history,
            Func<string, bool> isSpeakerAvailable, EncounterConditionRegistry conditions, EncounterConditionContext context, Func<float> random01)
        {
            if (Contains(pending, DialogueTrigger.AfterEvent))
            {
                var afterEvent = FirstForTrigger(DialogueTrigger.AfterEvent, dialogues, history, isSpeakerAvailable, conditions, context);
                if (afterEvent != null) return afterEvent;
            }
            foreach (var trigger in pending)
            {
                if (trigger == DialogueTrigger.AfterEvent) continue;
                var entry = FirstForTrigger(trigger, dialogues, history, isSpeakerAvailable, conditions, context);
                if (entry != null) return entry;
            }

            if (random01() >= RandomDialogueChance) return null;
            var randoms = EncounterCandidatePicker.FilterDialogues(dialogues, d => d.Occurrence == DialogueOccurrence.Random, history, isSpeakerAvailable, conditions, context);
            return EncounterCandidatePicker.PickWeighted(randoms, d => d.Weight, random01);
        }

        public static DialogueEntry PickManual(string characterId, IReadOnlyList<DialogueEntry> dialogues, IEncounterHistoryReader history,
            Func<string, bool> isSpeakerAvailable, EncounterConditionRegistry conditions, EncounterConditionContext context, Func<float> random01)
        {
            var manual = EncounterCandidatePicker.FilterDialogues(dialogues, d => d.Occurrence == DialogueOccurrence.Manual && Speaks(d, characterId), history, isSpeakerAvailable, conditions, context);
            if (manual.Count > 0) return manual[0];
            var repeat = EncounterCandidatePicker.FilterDialogues(dialogues, d => d.Occurrence == DialogueOccurrence.ManualRepeat && Speaks(d, characterId), history, isSpeakerAvailable, conditions, context);
            return EncounterCandidatePicker.PickWeighted(repeat, d => d.Weight, random01);
        }

        /// <summary>수동 걸기 목록의 "할 말 없음" 판정 - 목록을 열 때 상단원별로 부른다(설계 88 §4.1).</summary>
        public static bool HasManualTalk(string characterId, IReadOnlyList<DialogueEntry> dialogues, IEncounterHistoryReader history,
            Func<string, bool> isSpeakerAvailable, EncounterConditionRegistry conditions, EncounterConditionContext context)
            => PickManual(characterId, dialogues, history, isSpeakerAvailable, conditions, context, () => 0f) != null;

        private static DialogueEntry FirstForTrigger(DialogueTrigger trigger, IReadOnlyList<DialogueEntry> dialogues, IEncounterHistoryReader history,
            Func<string, bool> isSpeakerAvailable, EncounterConditionRegistry conditions, EncounterConditionContext context)
        {
            var candidates = EncounterCandidatePicker.FilterDialogues(dialogues, d => d.Occurrence == DialogueOccurrence.Trigger && d.Trigger == trigger, history, isSpeakerAvailable, conditions, context);
            return candidates.Count > 0 ? candidates[0] : null;
        }

        private static bool Speaks(DialogueEntry entry, string characterId)
        {
            if (entry.Speakers == null) return false;
            foreach (var s in entry.Speakers) if (s == characterId) return true;
            return false;
        }

        private static bool Contains(IReadOnlyList<DialogueTrigger> list, DialogueTrigger value)
        {
            foreach (var t in list) if (t == value) return true;
            return false;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    /// <summary>진행자가 쓰는 값·서비스 묶음. 테스트가 필요한 것만 채우도록 기본값을 둔다.</summary>
    public sealed class TripDialogueSources
    {
        public IEncounterContentReader Content;
        public IEncounterHistoryReader History;
        public IEncounterHistoryWriter HistoryWriter;
        public IDialogueScriptRunner Runner;
        public EncounterConditionRegistry Conditions = EncounterConditionRegistry.CreateDefault();
        public Func<EncounterConditionContext> Context = () => default;
        public Func<string, bool> IsHired = _ => true;
        public Func<IReadOnlyList<CrewMember>> Crew = () => Array.Empty<CrewMember>();
        public Func<string, string> NameOf = _ => null;
        public Func<float> Random01 = () => 0.99f;
        public Func<bool> IsHalted = () => false;
    }

    /// <summary>
    /// 상행 대화 진행 규칙(설계 88 §3·§4). Unity를 모르는 순수 C# - FieldManager 하위 껍데기(TripDialogueCoordinator)가 신호·프레임을 넘긴다.
    /// - 상황 신호는 다음 20초 검사까지 쌓아 두고, 검사 때 대화가 진행 중이면 그냥 버린다(소멸).
    /// - 상행 정지(전투·이벤트)·상행 종료 = 즉시 종료(기록 없음). 모달 화면(정비창·방향성 지시) = 일시정지.
    /// - 끝까지 진행한 대화는 종류와 무관하게 기록한다(조건 seen용 - 횟수 제한은 후보 거르기가 종류별로 판단).
    /// 대사 재생 상태(DialoguePlaybackController)는 대화마다 새로 만든다 - 이전 대화의 일시정지·선택지 상태가 남지 않게.
    /// </summary>
    public sealed class TripDialogueDirector : ITripDialogueStatus
    {
        private static readonly object ModalPauseReason = new();

        private sealed class Run
        {
            public DialogueEntry Entry;
            public DialoguePlaybackController Controller;
        }

        private readonly TripDialogueSources sources;
        private readonly SpeakerPresentation speakers;
        private readonly List<DialogueTrigger> pending = new();
        private ITripDialogueView dialogueView;
        private IManualTalkView manualView;
        private Run current;
        private bool modalOpen;

        public TripDialogueDirector(TripDialogueSources sources)
        {
            this.sources = sources;
            speakers = new SpeakerPresentation(sources.NameOf);
        }

        public bool IsRunning => current != null;
        public string CurrentDialogueId => current?.Entry.Id;
        public IReadOnlyList<DialogueTrigger> PendingTriggers => pending;

        /// <summary>이전 씬 화면으로 진행 중이던 대화는 화면을 건드리지 않고 끝낸다 - 그 화면은 씬과 함께 이미 파괴됐을 수 있다.</summary>
        public void BindViews(ITripDialogueView newDialogueView, IManualTalkView newManualView)
        {
            StopCurrent(hideView: false);
            if (dialogueView != null) dialogueView.Clicked -= Click;
            if (manualView != null)
            {
                manualView.OpenRequested -= HandleOpenRequested;
                manualView.MemberChosen -= HandleMemberChosen;
            }

            dialogueView = newDialogueView;
            manualView = newManualView;
            if (dialogueView != null) dialogueView.Clicked += Click;
            if (manualView != null)
            {
                manualView.OpenRequested += HandleOpenRequested;
                manualView.MemberChosen += HandleMemberChosen;
            }
            RefreshTalkAllowed();
        }

        public void OnMilestone(TripMilestone milestone)
        {
            switch (milestone)
            {
                case TripMilestone.TripStarted:
                    pending.Clear();
                    pending.Add(DialogueTrigger.Departure);
                    break;
                case TripMilestone.TripEnded:
                    StopCurrent(hideView: true);
                    pending.Clear();
                    manualView?.HideMembers();
                    break;
                default:
                    if (TripDialogueSelector.TryMapMilestone(milestone, out var trigger)) pending.Add(trigger);
                    break;
            }
        }

        public void OnDialogueCheck()
        {
            if (IsRunning || dialogueView == null)
            {
                pending.Clear();
                return;
            }
            var entry = TripDialogueSelector.PickForCheck(pending, sources.Content.Dialogues, sources.History, IsSpeakerAvailable, sources.Conditions, sources.Context(), sources.Random01);
            pending.Clear();
            if (entry != null) Start(entry);
        }

        public void OnHaltStarted()
        {
            StopCurrent(hideView: true);
            manualView?.HideMembers();
            RefreshTalkAllowed();
        }

        public void OnHaltEnded() => RefreshTalkAllowed();

        public void SetModalOpen(bool open)
        {
            modalOpen = open;
            current?.Controller.SetPaused(ModalPauseReason, open);
        }

        public void Tick(float deltaSeconds) => current?.Controller.Tick(deltaSeconds);

        public void Click() => current?.Controller.Click();

        public bool DebugStart(string dialogueId)
        {
            if (!sources.Content.TryGetDialogue(dialogueId, out var entry)) return false;
            StopCurrent(hideView: true);
            return Start(entry);
        }

        private bool Start(DialogueEntry entry)
        {
            if (dialogueView == null) return false;
            var controller = new DialoguePlaybackController(new StagedDialogueView(dialogueView, speakers), DialoguePlaybackOptions.Trip);
            if (modalOpen) controller.SetPaused(ModalPauseReason, true);
            var run = new Run { Entry = entry, Controller = controller };
            current = run;
            if (!sources.Runner.TryRun(entry.YarnNode, controller, completed => Finish(run, completed)))
            {
                if (current == run) current = null;
                return false;
            }
            manualView?.HideMembers();
            RefreshTalkAllowed();
            return true;
        }

        private void Finish(Run run, bool completed)
        {
            if (completed) sources.HistoryWriter?.RecordCompleted(run.Entry.Id);
            if (current != run) return;
            current = null;
            RefreshTalkAllowed();
        }

        private void StopCurrent(bool hideView)
        {
            if (current == null) return;
            var run = current;
            current = null;
            sources.Runner.Stop();
            if (hideView) run.Controller.Stop();
            RefreshTalkAllowed();
        }

        private void HandleOpenRequested()
        {
            if (!CanTalk) return;
            manualView.ShowMembers(BuildMembers());
        }

        private void HandleMemberChosen(string characterId)
        {
            manualView?.HideMembers();
            if (!CanTalk) return;
            var entry = TripDialogueSelector.PickManual(characterId, sources.Content.Dialogues, sources.History, IsSpeakerAvailable, sources.Conditions, sources.Context(), sources.Random01);
            if (entry != null) Start(entry);
        }

        private IReadOnlyList<ManualTalkMember> BuildMembers()
        {
            var context = sources.Context();
            var crew = sources.Crew();
            var list = new List<ManualTalkMember>(crew.Count);
            foreach (var member in crew)
            {
                var hasTalk = TripDialogueSelector.HasManualTalk(member.CharacterId, sources.Content.Dialogues, sources.History, IsSpeakerAvailable, sources.Conditions, context);
                list.Add(new ManualTalkMember(member.CharacterId, member.Name, hasTalk));
            }
            return list;
        }

        private bool CanTalk => dialogueView != null && manualView != null && !IsRunning && !sources.IsHalted();

        private void RefreshTalkAllowed() => manualView?.SetTalkAllowed(CanTalk);

        // 외부 인물은 고용 판정에서 제외한다(설계 87 §5.2).
        private bool IsSpeakerAvailable(string speakerId) => sources.Content.TryGetPerson(speakerId, out _) || sources.IsHired(speakerId);
    }
}
```

```csharp
    /// FieldManager 하위 컴포넌트 - 대화 진행 규칙(TripDialogueDirector)을 Bootstrap 수명에 붙인다(설계 88 §2). 의존성 조회·신호 구독·프레임 진행만 맡는다.
    /// 화면은 Field 로드마다 BindViews로 바뀐다. 씬이 언로드돼 화면이 파괴되면 다음 프레임에 연결을 끊는다 - 파괴된 화면에 그리지 않게.
    /// 대사 실행기·콘텐츠·이력 중 하나라도 없으면(Bootstrap 미재빌드) 대화 없이 상행만 진행한다.
    /// </summary>
    public class TripDialogueCoordinator : MonoBehaviour, ITripDialogueViewBinder, ITripDialogueDebug
    {
        private static readonly IReadOnlyList<DialogueTrigger> NoTriggers = new DialogueTrigger[0];

        private TripDialogueDirector director;
        private ITripCheckSignal checks;
        private ITripMilestoneSource milestones;
        private ITripHaltReader halt;
        private IUIVisibilitySignal visibility;
        private ITripDialogueView boundView;
        private IManualTalkView boundManual;

        // 판정 문맥 재료 - 판정 순간 값을 읽는다.
        private IDateReader date;
        private ITripItinerary itinerary;
        private IWorldMapReader worldMap;
        private ISessionState session;
        private IEncounterHistoryReader history;

        public bool IsRunning => director != null && director.IsRunning;
        public string CurrentDialogueId => director?.CurrentDialogueId;
        public IReadOnlyList<DialogueTrigger> PendingTriggers => director != null ? director.PendingTriggers : NoTriggers;

        public void Configure(IDependencyResolver registrar, ITripCheckSignal checks, ITripMilestoneSource milestones, ITripHaltReader halt)
        {
            Unsubscribe();
            director = null;

            registrar.TryResolve(out IEncounterContentReader content);
            registrar.TryResolve(out history);
            registrar.TryResolve(out IEncounterHistoryWriter historyWriter);
            registrar.TryResolve(out IDialogueScriptRunner runner);
            if (content == null || history == null || historyWriter == null || runner == null)
            {
                Debug.LogWarning($"{nameof(TripDialogueCoordinator)}: 콘텐츠·이력·대사 실행기 중 없는 것이 있어 상행 대화가 발생하지 않는다(Tools > Game > Build Bootstrap Scene).");
                return;
            }

            registrar.TryResolve(out IHiredCharacterRoster hired);
            registrar.TryResolve(out ICaravanRosterProvider roster);
            registrar.TryResolve(out ICharacterCatalogReader catalog);
            registrar.TryResolve(out date);
            registrar.TryResolve(out itinerary);
            registrar.TryResolve(out worldMap);
            registrar.TryResolve(out IUIManager uiManager);
            session = registrar.Resolve<ISessionState>();

            this.checks = checks;
            this.milestones = milestones;
            this.halt = halt;
            visibility = uiManager;

            director = new TripDialogueDirector(new TripDialogueSources
            {
                Content = content,
                History = history,
                HistoryWriter = historyWriter,
                Runner = runner,
                Context = BuildContext,
                IsHired = id => hired != null && hired.IsHired(id),
                Crew = () => CrewOf(roster),
                NameOf = id => NameOf(id, catalog, content),
                Random01 = () => Random.value,
                IsHalted = () => halt != null && halt.IsHalted,
            });

            if (checks != null) checks.DialogueCheck += director.OnDialogueCheck;
            if (milestones != null) milestones.Raised += director.OnMilestone;
            if (halt != null)
            {
                halt.HaltStarted += director.OnHaltStarted;
                halt.HaltEnded += director.OnHaltEnded;
            }
            if (visibility != null)
            {
                visibility.ModalPopupOpenChanged += director.SetModalOpen;
                director.SetModalOpen(visibility.IsModalPopupOpen);
            }
            director.BindViews(boundView, boundManual);
        }

        public void BindViews(ITripDialogueView dialogueView, IManualTalkView manualTalkView)
        {
            boundView = dialogueView;
            boundManual = manualTalkView;
            director?.BindViews(dialogueView, manualTalkView);
        }

        public bool DebugStart(string dialogueId) => director != null && director.DebugStart(dialogueId);

        private void Update()
        {
            if (director == null) return;
            if (boundView is Object view && view == null)
            {
                BindViews(null, null);
                return;
            }
            director.Tick(Time.deltaTime);
        }

        private EncounterConditionContext BuildContext()
        {
            var hasLeg = itinerary != null && itinerary.HasItinerary;
            var today = date != null ? date.Today : default;
            return EncounterConditionContextFactory.Create(today, hasLeg, hasLeg ? itinerary.CurrentLeg : default, RegionOf, session != null ? session.Progress : 0f, history);
        }

        private int RegionOf(int cityId) => worldMap != null && worldMap.TryGetCity(cityId, out var city) ? city.RegionId : -1;

        private static IReadOnlyList<CrewMember> CrewOf(ICaravanRosterProvider roster)
        {
            var list = new List<CrewMember>();
            if (roster == null) return list;
            foreach (var unit in roster.GetRoster())
            {
                if (unit.Kind == FormationUnitKind.Character) list.Add(new CrewMember(unit.Id, unit.DisplayName));
            }
            return list;
        }

        private static string NameOf(string id, ICharacterCatalogReader catalog, IEncounterContentReader content)
        {
            if (catalog != null && catalog.TryGet(id, out var profile)) return profile.Name;
            return content.TryGetPerson(id, out var person) ? person.Name : null;
        }

        private void Unsubscribe()
        {
            if (director == null) return;
            if (checks != null) checks.DialogueCheck -= director.OnDialogueCheck;
            if (milestones != null) milestones.Raised -= director.OnMilestone;
            if (halt != null)
            {
                halt.HaltStarted -= director.OnHaltStarted;
                halt.HaltEnded -= director.OnHaltEnded;
            }
            if (visibility != null) visibility.ModalPopupOpenChanged -= director.SetModalOpen;
        }

        private void OnDestroy() => Unsubscribe();
    }
}
```

```csharp
    foreach (var r in ws.RowsUsed()) existing.Add(r.Cell(1).GetString());

        added++;
    }
    wb.Save();
}
```

```csharp
using System;
using System.Collections.Generic;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace Game.Core
{
    /// <summary>
    /// Field 말풍선 + 좌우 스탠딩(설계 88 §5). 표시 전용 - 자리·밝기·이름은 StagedDialogueView가 정해 준다. IUIPanel이 아니다(상행을 막지 않고 이동 뷰 위에 겹칠 뿐,
    /// FieldEncounterWarningView와 같은 성격). 루트는 항상 켜 두고 Body만 켜고 끈다 - 진행자가 루트 컴포넌트를 계속 참조하기 때문이다.
    /// 선택지 줄은 템플릿에서 처음 필요할 때만 만들고 이후 재사용한다.
    /// </summary>
    public class FieldDialogueView : MonoBehaviour, ITripDialogueView
    {
        private const float DimFactor = 0.45f;
        private const float LockedAlpha = 0.5f;

        [SerializeField] private GameObject body;
        [SerializeField] private FieldDialogueClickCatcher clickCatcher;
        [SerializeField] private TMP_Text speakerNameText;
        [SerializeField] private TMP_Text lineText;
        [SerializeField] private GameObject continueMarker;
        [SerializeField] private GameObject leftRoot;
        [SerializeField] private Image leftFigure;
        [SerializeField] private TMP_Text leftName;
        [SerializeField] private GameObject rightRoot;
        [SerializeField] private Image rightFigure;
        [SerializeField] private TMP_Text rightName;
        [SerializeField] private GameObject optionsRoot;
        [SerializeField] private Button optionTemplate;
        [SerializeField] private Image timerFill;

        private sealed class OptionRow
        {
            public Button Button;
            public TMP_Text Label;
            public CanvasGroup Group;
        }

        private readonly List<OptionRow> rows = new();

        public event Action Clicked;

        private void Awake()
        {
            if (clickCatcher != null) clickCatcher.Clicked += RaiseClicked;
            SetVisible(false);
        }

        private void OnDestroy()
        {
            if (clickCatcher != null) clickCatcher.Clicked -= RaiseClicked;
        }

        private void RaiseClicked() => Clicked?.Invoke();

        public void SetVisible(bool visible)
        {
            body.SetActive(visible);
            if (clickCatcher != null) clickCatcher.gameObject.SetActive(visible);
            if (!visible) HideOptions();
        }

        public void SetBubble(string speakerName, string text)
        {
            speakerNameText.text = speakerName;
            lineText.text = text;
            lineText.maxVisibleCharacters = 0;
        }

        public void SetVisibleCharacters(int count) => lineText.maxVisibleCharacters = count;

        public void SetContinueMarker(bool shown) => continueMarker.SetActive(shown);

        public void SetStanding(DialogueSide side, StandingState state)
        {
            var root = side == DialogueSide.Left ? leftRoot : rightRoot;
            var figure = side == DialogueSide.Left ? leftFigure : rightFigure;
            var nameText = side == DialogueSide.Left ? leftName : rightName;
            root.SetActive(state.Visible);
            if (!state.Visible) return;
            var color = state.Highlighted ? state.Color : state.Color * DimFactor;
            color.a = 1f;
            figure.color = color;
            nameText.text = state.Name;
        }

        public void ShowOptions(IReadOnlyList<DialogueOptionData> options, Action<int> onChosen)
        {
            optionsRoot.SetActive(true);
            for (var i = 0; i < options.Count; i++)
            {
                var row = GetRow(i);
                var option = options[i];
                row.Button.gameObject.SetActive(true);
                row.Button.interactable = option.IsAvailable;
                row.Group.alpha = option.IsAvailable ? 1f : LockedAlpha;
                row.Label.text = option.IsAvailable || option.LockText.Length == 0 ? option.Text : $"{option.Text} ({option.LockText})";
                row.Button.onClick.RemoveAllListeners();
                var index = option.Index;
                row.Button.onClick.AddListener(() => onChosen(index));
            }
            for (var i = options.Count; i < rows.Count; i++) rows[i].Button.gameObject.SetActive(false);
            timerFill.gameObject.SetActive(false);   // 제한시간이 있으면 진행자가 곧바로 SetOptionTimeRemaining을 부른다
        }

        public void SetOptionTimeRemaining(float fraction01)
        {
            timerFill.gameObject.SetActive(true);
            timerFill.fillAmount = fraction01;
        }

        public void HideOptions()
        {
            optionsRoot.SetActive(false);
            timerFill.gameObject.SetActive(false);
        }

        private OptionRow GetRow(int index)
        {
            while (rows.Count <= index)
            {
                var button = Instantiate(optionTemplate, optionTemplate.transform.parent);
                rows.Add(new OptionRow
                {
                    Button = button,
                    Label = button.GetComponentInChildren<TMP_Text>(true),
                    Group = button.GetComponent<CanvasGroup>(),
                });
            }
            return rows[index];
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace Game.Core
{
    /// <summary>
    /// "대화" 버튼 + 상단원 목록(설계 88 §6). 표시 전용 - 모달이 아니어서 열려 있어도 상행은 계속된다. 버튼 활성 = 상단 버튼 조건(화면 공개·인카운터, FieldUIController)
    /// × 대화 조건(대화 중·상행 정지 아님, 진행자). 버튼이 꺼지면 목록도 닫는다. 목록 줄은 템플릿에서 처음 필요할 때만 만들고 재사용한다.
    /// </summary>
    public class FieldManualTalkView : MonoBehaviour, IManualTalkView
    {
        private const float NoTalkAlpha = 0.5f;
        private const string NoTalkSuffix = " - 할 말 없음";

        [SerializeField] private Button talkButton;
        [SerializeField] private GameObject memberList;
        [SerializeField] private Button memberTemplate;

        private sealed class MemberRow
        {
            public Button Button;
            public TMP_Text Label;
            public CanvasGroup Group;
        }

        private readonly List<MemberRow> rows = new();
        private bool topLevelAllowed = true;
        private bool talkAllowed;

        public event Action OpenRequested;
        public event Action<string> MemberChosen;

        private void Awake()
        {
            talkButton.onClick.AddListener(HandleTalkClicked);
            memberList.SetActive(false);
            Apply();
        }

        public void SetTopLevelAllowed(bool allowed)
        {
            topLevelAllowed = allowed;
            Apply();
        }

        public void SetTalkAllowed(bool allowed)
        {
            talkAllowed = allowed;
            Apply();
        }

        public void ShowMembers(IReadOnlyList<ManualTalkMember> members)
        {
            for (var i = 0; i < members.Count; i++)
            {
                var row = GetRow(i);
                var member = members[i];
                row.Button.gameObject.SetActive(true);
                row.Button.interactable = member.HasTalk;
                row.Group.alpha = member.HasTalk ? 1f : NoTalkAlpha;
                row.Label.text = member.HasTalk ? member.Name : member.Name + NoTalkSuffix;
                row.Button.onClick.RemoveAllListeners();
                var id = member.CharacterId;
                row.Button.onClick.AddListener(() => MemberChosen?.Invoke(id));
            }
            for (var i = members.Count; i < rows.Count; i++) rows[i].Button.gameObject.SetActive(false);
            memberList.SetActive(true);
        }

        public void HideMembers() => memberList.SetActive(false);

        private void HandleTalkClicked()
        {
            if (memberList.activeSelf) HideMembers();
            else OpenRequested?.Invoke();
        }

        private void Apply()
        {
            var on = topLevelAllowed && talkAllowed;
            talkButton.interactable = on;
            if (!on) HideMembers();
        }

        private MemberRow GetRow(int index)
        {
            while (rows.Count <= index)
            {
                var button = Instantiate(memberTemplate, memberTemplate.transform.parent);
                rows.Add(new MemberRow
                {
```

```csharp
            if (dialogue != null && content != null)
            {
                foreach (var entry in content.Dialogues)
                {
                    if (GUILayout.Button($"{entry.Id} ({entry.Occurrence})")) dialogue.DebugStart(entry.Id);
                }
            }
            GUILayout.EndScrollView();
            GUILayout.EndArea();
        }

        private string DescribeStatus()
        {
            status.Clear();
            if (dialogue == null) return "대화 진행자 없음(Build Bootstrap Scene)";
            if (checks != null) status.AppendLine($"다음 대화 검사 {checks.SecondsUntil(TripCheckKind.Dialogue):0.0}s");
            status.AppendLine("진행 중: " + (dialogue.IsRunning ? dialogue.CurrentDialogueId : "없음"));
            status.Append("쌓인 일: ");
            for (var i = 0; i < dialogue.PendingTriggers.Count; i++) status.Append(i == 0 ? "" : ", ").Append(dialogue.PendingTriggers[i]);
            return status.ToString();
        }
```

```csharp
        [MenuItem("Tools/Game/Debug/Install/Dialogue Debug")]
        public static void Install()
        {

            if (fieldManager == null)
            {

            if (component != null) Undo.DestroyObjectImmediate(component);
```
