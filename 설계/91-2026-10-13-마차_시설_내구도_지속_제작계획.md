# 마차·시설 내구도 지속 제작 계획 (묶음 A~C)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 마차·시설 개체별 내구도를 상행을 넘어 저장하고(테이블 최대치), 전투 시작·종료에 연결하고(시설 잔해·파괴 제거 포함), 마구간에서 개별·전체 수리를 할 수 있게 한다.

**Architecture:** 값 보관 규칙은 Unity를 모르는 순수 C#(`CaravanAssetDurabilityLedger`·`CaravanAssetDurabilityRules`·`CaravanAssetRepairService`·`CaravanAssetDestroyer`·`StableOwnedListModel`)에 두고 EditMode로 검증한다. Bootstrap 상주 껍데기 `CaravanAssetDurabilityManager`가 원장을 DI 수명에 붙인다. 전투는 결과(`BattleResult`)에 보호 유닛 기록을 싣고, 정산(`BattleAftermathApplier`)이 내구도 저장과 공용 파괴 처리를 한 번에 한다. 마구간 보유 영역은 `StableOwnedSection`이 맡고 `StablePanel`은 구매만 남긴다.

**Tech Stack:** Unity 6000.3, C#, NUnit(EditMode), TextMeshPro, ClosedXML(에디터, 엑셀 열 추가), Unity CLI(`ut.sh`, `unity command ...`)

**Spec:** `Docs/설계/91-2026-10-13-마차_시설_내구도_지속_아키텍처.md` (기획: `Docs/기획/시스템/90-2026-10-13-마차_시설_내구도_지속_기획.md`)

**범위:** 설계 91 §9 묶음 A·B·C. 묶음 D(이벤트 효과 `<<durability>>`·대상 선정 정책·화물 이동·궤주 공용 진입점)는 설계 89 제작 후 별도 계획으로 만든다 — 대상 선정 정책 클래스(설계 91 §3.3 첫 줄)도 D의 소비자와 함께 만든다.

```csharp
var root = System.IO.Path.Combine(UnityEngine.Application.dataPath, "Table/Caravan");
var log = new System.Text.StringBuilder();
foreach (var name in new[] { "Wagon", "Facility" })
{
    var path = System.IO.Path.Combine(root, name + ".xlsx");
    using (var wb = new ClosedXML.Excel.XLWorkbook(path))
    {
        var ws = wb.Worksheet(name);
        var lastCol = ws.Row(1).LastCellUsed().Address.ColumnNumber;
        var col = -1;
        for (var c = 1; c <= lastCol; c++) if (ws.Cell(1, c).GetString() == "MaxDurability") col = c;
        if (col < 0) { col = lastCol + 1; ws.Cell(1, col).Value = "MaxDurability"; }
        var filled = 0;
        var lastRow = ws.LastRowUsed().RowNumber();
        for (var r = 2; r <= lastRow; r++)
        {
            if (ws.Cell(r, 1).GetString().Length == 0) continue;
            if (ws.Cell(r, col).GetString().Length > 0) continue;
            ws.Cell(r, col).Value = 200;
            filled++;
        }
        wb.Save();
        log.Append(name).Append(" col=").Append(col).Append(" filled=").Append(filled).Append("; ");
    }
}
return log.ToString();
```

```csharp
using System;

namespace Game.Core
{
    /// <summary>
    /// 마차·시설 내구도의 표시·손상·수리비 규칙(Docs/기획/시스템/90번 §3-9·§4-4·§4-15, 설계 91 §3.3). 마구간 목록·개별 정보·디버그·이벤트 결과 줄이
    /// 같은 숫자를 보여야 해서 한 곳에 둔다. 손상 판정을 표시값 기준으로 하는 이유: "200/200"으로 보이는 개체에 수리비가 붙거나 색이 바뀌지 않게.
    /// </summary>
    public static class CaravanAssetDurabilityRules
    {
        // 수리비 계수(기획 90번 §3-9) - 종류별 계수 열이 생기면 CoefficientOf만 바꾼다.
        public const float DefaultRepairCoefficient = 0.5f;
        // 부동소수 오차(예: 50.0000001)가 올림에서 1골드로 커지지 않게 이 자릿수에서 먼저 반올림한다.
        private const int RoundingDigits = 4;

        /// <summary>표시값 - 정수 올림(0.4 → 1). 살아 있는 개체가 0으로 보이지 않게 한다(기획 90번 §4-4).</summary>
        public static int Display(float value) => (int)Math.Ceiling(Math.Round((double)value, RoundingDigits));

        public static bool IsDamaged(float current, int max) => Display(current) < max;

        public static float CoefficientOf(CaravanAssetProfile profile) => DefaultRepairCoefficient;

        /// <summary>개별 수리비 = 올림(가격 × 깎인 비율 × 계수). 손상 없으면 0(기획 90번 §3-9·§4-15).</summary>
        public static int RepairCost(int price, float current, int max, float coefficient)
        {
            if (max <= 0 || !IsDamaged(current, max)) return 0;
            var missing = Math.Min(Math.Max((double)max - current, 0d), max);
            var raw = price * (missing / max) * coefficient;
            return (int)Math.Ceiling(Math.Round(raw, RoundingDigits));
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    /// <summary>
    /// 개체 Id → 현재 내구도(Docs/설계/91번 §3.2). 보유 목록 변경을 구독해 새 개체는 최대치로 등록하고 사라진 개체는 지운다 - 구매·시작 보유·
    /// 파괴 제거가 저마다 내구도를 챙기지 않아도 되게. 최대치는 저장하지 않고 조회 때마다 종류 정보에서 읽는다 - 테이블 값이 바뀌어도
    /// 새 최대치로 잘려 보이게(기획 90번 §4-5). MonoBehaviour와 분리한 이유는 Unity 밖에서 규칙을 검증하기 위해서다.
    /// </summary>
    public sealed class CaravanAssetDurabilityLedger : ICaravanAssetDurabilityReader, ICaravanAssetDurabilityWriter, IDisposable
    {
        private static readonly FormationUnitKind[] Kinds = { FormationUnitKind.Wagon, FormationUnitKind.Facility };

        private readonly IOwnedCaravanAssetReader owned;
        private readonly Dictionary<string, float> currentById = new();
        private readonly HashSet<string> ownedIds = new();
        private readonly List<string> staleIds = new();

        public event Action Changed;

        public CaravanAssetDurabilityLedger(IOwnedCaravanAssetReader owned)
        {
            this.owned = owned;
            owned.OnOwnedChanged += Sync;
            Sync();
        }

        public void Dispose() => owned.OnOwnedChanged -= Sync;

        public bool TryGet(string instanceId, out float current, out int max)
        {
            current = 0f;
            max = 0;
            if (instanceId == null || !currentById.TryGetValue(instanceId, out var stored) || !owned.TryGetOwned(instanceId, out var asset)) return false;

            max = asset.Profile.MaxDurability;
            current = Math.Min(stored, max);
            return true;
        }

        public bool TrySet(string instanceId, float value)
        {
            if (!TryGet(instanceId, out var current, out var max)) return false;

            var clamped = Math.Max(0f, Math.Min(value, max));
            // 저장값이 최대치보다 큰 채(최대치 축소) 같은 값을 쓰면 저장값만 맞추고 알리지 않는다 - 보이는 값은 그대로다.
            var visibleChange = clamped != current;
            currentById[instanceId] = clamped;
            if (visibleChange) Changed?.Invoke();
            return true;
        }

        public bool TryRestore(string instanceId) => TryGet(instanceId, out _, out var max) && TrySet(instanceId, max);

        private void Sync()
        {
            var changed = false;
            ownedIds.Clear();
            foreach (var kind in Kinds)
            {
                foreach (var id in owned.GetOwnedIds(kind))
                {
                    ownedIds.Add(id);
                    if (currentById.ContainsKey(id) || !owned.TryGetOwned(id, out var asset)) continue;
                    currentById[id] = asset.Profile.MaxDurability;
                    changed = true;
                }
            }

            staleIds.Clear();
            foreach (var id in currentById.Keys)
            {
                if (!ownedIds.Contains(id)) staleIds.Add(id);
            }
            foreach (var id in staleIds) currentById.Remove(id);

            if (changed || staleIds.Count > 0) Changed?.Invoke();
        }
    }
}
```

```csharp
    /// <summary>
    /// 마구간 수리 판정·실행(Docs/설계/91번 §5.1, 기획 90번 §3-10·§4.3). 구매 서비스(CaravanAssetPurchaseService)와 같이 판정과 실행을 나눠
    /// 버튼 상태와 실제 처리가 같은 기준을 쓰게 한다. 지출 → 복원 순이고 복원이 실패하면 환급한다.
    /// 전체 수리는 합계를 한 번에 지출한다 - 합계가 부족하면 아무것도 고치지 않는다(부분 수리 없음, §3-10).
    /// 지출 순서(개인 → 적재 골드 상자, 기획 82번)는 IGoldSpender가 처리한다.
    /// </summary>
    public sealed class CaravanAssetRepairService
    {
        private static readonly FormationUnitKind[] Kinds = { FormationUnitKind.Wagon, FormationUnitKind.Facility };

        private readonly IGoldSpender gold;
        private readonly IOwnedCaravanAssetReader owned;
        private readonly ICaravanAssetDurabilityReader durability;
        private readonly ICaravanAssetDurabilityWriter writer;

        public CaravanAssetRepairService(IGoldSpender gold, IOwnedCaravanAssetReader owned, ICaravanAssetDurabilityReader durability, ICaravanAssetDurabilityWriter writer)
        {
            this.gold = gold;
            this.owned = owned;
            this.durability = durability;
            this.writer = writer;
        }

        public int CostOf(string instanceId)
        {
            if (!owned.TryGetOwned(instanceId, out var asset) || !durability.TryGet(instanceId, out var current, out var max)) return 0;
            return CaravanAssetDurabilityRules.RepairCost(asset.Profile.Price, current, max, CaravanAssetDurabilityRules.CoefficientOf(asset.Profile));
        }

        public int TotalCost()
        {
            var total = 0;
            foreach (var (_, cost) in DamagedTargets()) total += cost;
            return total;
        }

        public CaravanAssetRepairCheck Evaluate(string instanceId) => Check(CostOf(instanceId));

        public CaravanAssetRepairCheck EvaluateAll() => Check(TotalCost());

        public bool TryRepair(string instanceId)
        {
            var cost = CostOf(instanceId);
            if (Check(cost) != CaravanAssetRepairCheck.Available || !gold.TrySpend(cost)) return false;
            if (writer.TryRestore(instanceId)) return true;

            gold.Refund(cost); // 복원 실패 롤백(구매의 발급 실패 롤백과 같음)
            return false;
        }

        public bool TryRepairAll()
        {
            var targets = DamagedTargets();
            var total = 0;
            foreach (var (_, cost) in targets) total += cost;
            if (Check(total) != CaravanAssetRepairCheck.Available || !gold.TrySpend(total)) return false;

            var failed = 0;
            foreach (var (id, cost) in targets)
            {
                if (!writer.TryRestore(id)) failed += cost;
            }
            if (failed > 0) gold.Refund(failed);
            return failed < total;
        }

        private CaravanAssetRepairCheck Check(int cost)
        {
            if (cost <= 0) return CaravanAssetRepairCheck.NotDamaged;
            return gold.CanAfford(cost) ? CaravanAssetRepairCheck.Available : CaravanAssetRepairCheck.InsufficientFunds;
        }

        // 보유 순서(마차 → 시설) - 전체 수리 합계와 개별 비용이 같은 계산을 거치게 한다(기획 90번 §4-15).
        private List<(string id, int cost)> DamagedTargets()
        {
            var targets = new List<(string, int)>();
            foreach (var kind in Kinds)
            {
                foreach (var id in owned.GetOwnedIds(kind))
                {
                    var cost = CostOf(id);
                    if (cost > 0) targets.Add((id, cost));
                }
            }
            return targets;
        }
    }
}
```

```csharp
namespace Game.Core
```

```csharp
    public readonly struct CaravanAssetDestructionResult
    {
        public int Relocated { get; }
        public int Released { get; }
        /// <summary>정리 후 마차 덩어리가 2개 이상 - 대열 정리 단계 대상.</summary>
        public bool Disconnected { get; }

        public CaravanAssetDestructionResult(int relocated, int released, bool disconnected)

            Relocated = relocated;

            Disconnected = disconnected;
        }

    /// <summary>
    /// 파괴된 마차·시설을 대열에서 빼고(재배치·활동 취소) 보유 목록에서 제거한다(Docs/설계/91번 §4.4). 전투 정산과 이벤트 효과(설계 91번 §6.2)가
    /// 같은 순서를 쓰도록 정산기에서 뽑았다. 순서 제약: 대열 조회는 보유 제거 **전에** 만든다 - 뺀 뒤엔 파괴 개체를 마차·시설로 인식하지 못해
    /// 그 칸의 영역을 계산하지 못한다. 의존성은 모두 선택적이다 - 없는 쪽 처리만 건너뛴다(배틀 테스트 씬 등).
    /// 내구도 기록은 보유 변경 알림으로 원장이 지우므로 여기서 다루지 않는다.
    /// </summary>
    public sealed class CaravanAssetDestroyer
    {
        private readonly IFormationRepository formation;
        private readonly ICaravanRosterProvider roster;
        private readonly IFieldFormationActivityRepository activities;
        private readonly IOwnedCaravanAssetRemover remover;

        public CaravanAssetDestroyer(IFormationRepository formation, ICaravanRosterProvider roster, IFieldFormationActivityRepository activities, IOwnedCaravanAssetRemover remover)
        {
            this.formation = formation;
            this.roster = roster;
            this.activities = activities;
            this.remover = remover;
        }

        public CaravanAssetDestructionResult Destroy(IReadOnlyCollection<string> ids)
        {
            ids ??= Array.Empty<string>();
            int relocated = 0, released = 0;
            var disconnected = false;

            if (formation != null && formation.TryLoadCurrent(out var layout) && layout != null)
            {
                var lookup = FormationAreaRules.LookupFrom(roster);
                if (ids.Count > 0)
                {
                    IReadOnlyList<FormationActivity> active = activities != null ? activities.ActiveActivities : Array.Empty<FormationActivity>();
                    var relocation = FormationWreckRelocator.Relocate(layout, ids, lookup, active);
                    formation.Apply(relocation.Layout);

                    if (activities != null)
                    {
                        foreach (var unitId in relocation.CancelledActivityUnitIds)
                        {
                            if (activities.IsUnitBusy(unitId)) activities.Cancel(unitId);
                        }
                    }

                    relocated = relocation.Relocated.Count;
                    released = relocation.Released.Count;
                    disconnected = relocation.ComponentCount > 1;
                }
                else
                {
                    // 파괴가 없어도 이미 끊어진 대열(이전 궤주 뒤 등)이면 대열 정리 단계 대상이다.
                    disconnected = !FormationArea.Compute(layout, lookup).WagonsConnected;
                }
            }

            // 보유 목록 제거 - 물류품 섹션·내구도 기록도 보유 변경 알림으로 함께 사라진다.
            if (remover != null)
            {
                foreach (var id in ids) remover.TryRemoveOwned(id);
            }

            return new CaravanAssetDestructionResult(relocated, released, disconnected);
        }
    }
}
```

```csharp
using System.Collections.Generic;

namespace Game.Core
{
    /// <summary>마구간 보유 목록 한 줄 - 종류 제목("마차 2") 또는 개체("1번 마차이름1  150/200").</summary>

    {
        public bool IsHeader { get; }
        public string InstanceId { get; }
        public string Name { get; }
        public string DurabilityText { get; }
        public bool Damaged { get; }

        public StableOwnedEntry(bool isHeader, string instanceId, string name, string durabilityText, bool damaged)
        {
            IsHeader = isHeader;
            InstanceId = instanceId;
            Name = name;
            DurabilityText = durabilityText;
            Damaged = damaged;
        }
    }

    /// <summary>
    /// 마구간 보유 목록 줄 만들기(Docs/설계/91번 §5.2, 기획 90번 §4-12). 화면(TMP)과 분리해 줄 구성·표시값을 Unity 밖에서 검증한다.
    /// 종류 순서는 마차 → 시설(정비창 팔레트 순서와 같음), 종류 안은 보유 순서. 제목의 상한 표기 없음은 기획 80번 §3-10 그대로.
    /// </summary>
    public static class StableOwnedListModel
    {
        public const string MissingText = "값 없음";
        private static readonly FormationUnitKind[] Kinds = { FormationUnitKind.Wagon, FormationUnitKind.Facility };

        public static List<StableOwnedEntry> Build(IOwnedCaravanAssetReader owned, ICaravanAssetCatalogReader catalog, ICaravanAssetDurabilityReader durability)
        {
            var entries = new List<StableOwnedEntry>();
            foreach (var kind in Kinds)
            {
                var ids = owned.GetOwnedIds(kind);
                entries.Add(new StableOwnedEntry(true, null, $"{catalog.GetKindLabel(kind)} {ids.Count}", string.Empty, false));
                foreach (var id in ids)
                {
                    if (!owned.TryGetOwned(id, out var asset)) continue;
                    var text = Describe(durability, id, out var damaged);
                    entries.Add(new StableOwnedEntry(false, id, OwnedCaravanAssetNames.Format(asset), text, damaged));
                }
            }
            return entries;
        }

        /// <summary>"현재/최대"(현재는 올림). 내구도 저장소가 없거나 기록이 없으면 "값 없음".</summary>
        public static string Describe(ICaravanAssetDurabilityReader durability, string instanceId, out bool damaged)
        {
            damaged = false;
            if (durability == null || !durability.TryGet(instanceId, out var current, out var max)) return MissingText;
            damaged = CaravanAssetDurabilityRules.IsDamaged(current, max);
            return $"{CaravanAssetDurabilityRules.Display(current)}/{max}";
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;

namespace Game.Core
{
    /// <summary>
    /// 마구간 보유 영역(Docs/설계/91번 §5.2, 기획 90번 §3-12·§3-13·§4.3). 목록 화면과 개별 정보 화면을 같은 자리에서 하나만 켠다.
    /// StablePanel에서 분리한 이유: 구매와 수리 두 흐름을 한 패널이 같이 갖지 않게(SRP). plain C#이며 HubUIWiring이 Hub 로드마다 새로 만든다 -
    /// 골드·보유·내구도 이벤트를 구독하므로 StablePanel.Dispose가 함께 해제한다. 수리 서비스·내구도 저장소가 없으면 "값 없음" + 수리 비활성.
    /// </summary>
    public sealed class StableOwnedSection : IDisposable
    {
        private const int VisibleRows = 10; // 목록 영역에 한 번에 보이는 줄 수(잠정)
        private const string NotNeededText = "수리할 필요 없음";
        private const string InsufficientText = "골드 부족";
        private const string RepairAllText = "전체 수리";

        private readonly StableOwnedElements elements;
        private readonly IOwnedCaravanAssetReader owned;
        private readonly ICaravanAssetCatalogReader catalog;
        private readonly ICaravanAssetDurabilityReader durability; // 선택적
        private readonly CaravanAssetRepairService repair;          // 선택적
        private readonly IGoldSpender gold;
        private readonly List<StableOwnedRowView> rows = new();

        private string selectedId;
        private bool isOpen;

        public StableOwnedSection(StableOwnedElements elements, IOwnedCaravanAssetReader owned, ICaravanAssetCatalogReader catalog, ICaravanAssetDurabilityReader durability, CaravanAssetRepairService repair, IGoldSpender gold)
        {
            this.elements = elements;
            this.owned = owned;
            this.catalog = catalog;
            this.durability = durability;
            this.repair = repair;
            this.gold = gold;

            elements.RowTemplate.gameObject.SetActive(false);
            elements.RepairAllButton.onClick.RemoveAllListeners();
            elements.RepairAllButton.onClick.AddListener(RepairAll);
            elements.RepairButton.onClick.RemoveAllListeners();
            elements.RepairButton.onClick.AddListener(RepairSelected);
            elements.BackButton.onClick.RemoveAllListeners();
            elements.BackButton.onClick.AddListener(ShowList);

            gold.Changed += HandleChanged;
            owned.OnOwnedChanged += HandleChanged;
            if (durability != null) durability.Changed += HandleChanged;
        }

        // 열 때마다 목록 화면부터(설계 91번 §5.2).
        public void Open()
        {
            isOpen = true;
            selectedId = null;
            Refresh();
        }

        public void Close() => isOpen = false;

        public void Dispose()
        {
            gold.Changed -= HandleChanged;
            owned.OnOwnedChanged -= HandleChanged;
            if (durability != null) durability.Changed -= HandleChanged;
        }

        private void HandleChanged()
        {
            if (isOpen) Refresh();
        }

        private void Refresh()
        {
            // 보던 개체가 사라졌으면 목록으로 돌아간다.
            if (selectedId != null && !owned.TryGetOwned(selectedId, out _)) selectedId = null;

            var showDetail = selectedId != null;
            elements.ListRoot.SetActive(!showDetail);
            elements.DetailRoot.SetActive(showDetail);
            if (showDetail) RenderDetail();
            else RenderList();
        }

        private void RenderList()
        {
            var entries = StableOwnedListModel.Build(owned, catalog, durability);
            var rowHeight = Mathf.Max(elements.ListViewport.rect.height, 1f) / VisibleRows;

            for (var i = 0; i < entries.Count; i++)
            {
                if (i >= rows.Count) rows.Add(UnityEngine.Object.Instantiate(elements.RowTemplate, elements.ListContent));

                var row = rows[i];
                row.gameObject.SetActive(true);
                var entry = entries[i];
                if (entry.IsHeader) row.BindHeader(entry.Name);
                else row.BindAsset(entry.Name, entry.DurabilityText, entry.Damaged, () => Select(entry.InstanceId));

                var rect = row.RectTransform;
                rect.anchorMin = new Vector2(0f, 1f);
                rect.anchorMax = new Vector2(1f, 1f);
                rect.pivot = new Vector2(0.5f, 1f);
                rect.anchoredPosition = new Vector2(0f, -i * rowHeight);
                rect.sizeDelta = new Vector2(0f, rowHeight);
            }
            for (var i = entries.Count; i < rows.Count; i++) rows[i].gameObject.SetActive(false);
            elements.ListContent.sizeDelta = new Vector2(elements.ListContent.sizeDelta.x, entries.Count * rowHeight);

            var total = repair?.TotalCost() ?? 0;
            var check = repair?.EvaluateAll() ?? CaravanAssetRepairCheck.NotDamaged;
            elements.RepairAllLabel.text = total > 0 ? $"{RepairAllText} ({total:N0})" : RepairAllText;
            elements.RepairAllButton.interactable = check == CaravanAssetRepairCheck.Available;
            // 손상 없음은 사유 없이 비활성(기획 90번 §4-16).
            elements.RepairAllReason.text = check == CaravanAssetRepairCheck.InsufficientFunds ? InsufficientText : string.Empty;
        }

        private void RenderDetail()
        {
            owned.TryGetOwned(selectedId, out var asset);
            elements.DetailName.text = OwnedCaravanAssetNames.Format(asset);
            elements.DetailKind.text = asset.Profile.KindLabel;
            elements.DetailDurability.text = "내구도 " + StableOwnedListModel.Describe(durability, selectedId, out _);
            elements.DetailCost.text = repair != null ? $"수리비 {repair.CostOf(selectedId):N0}" : "수리비 " + StableOwnedListModel.MissingText;

            var check = repair?.Evaluate(selectedId);
            elements.RepairButton.interactable = check == CaravanAssetRepairCheck.Available;
            elements.DetailReason.text = check switch
            {
                CaravanAssetRepairCheck.NotDamaged => NotNeededText,
                CaravanAssetRepairCheck.InsufficientFunds => InsufficientText,
                _ => string.Empty,
            };
        }

        private void Select(string instanceId)
        {
            selectedId = instanceId;
            Refresh();
        }

        private void ShowList()
        {
            selectedId = null;
            Refresh();
        }

        // 성공하면 골드·내구도 이벤트로 다시 그려지지만, 실패(경합)도 버튼 상태를 맞추도록 한 번 더 그린다. 확인 창 없음(기획 90번 §4-18).
        private void RepairSelected()
        {
            if (selectedId != null) repair?.TryRepair(selectedId);
            Refresh();
        }

        private void RepairAll()
        {
            repair?.TryRepairAll();
            Refresh();
        }
    }
}
```

```csharp
        // 배치 기준값 - Task 9 실측으로 확정한다(기존 UI·디버그 UI와 겹치지 않는 자리).
        private const float ToggleWidth = 90f;
        private const float ToggleHeight = 26f;
        private const float Margin = 8f;
        private const float PanelWidth = 380f;
        private const float PanelHeight = 260f;
        private const float StepRatio = 0.1f;
        private const float NearZeroRatio = 0.01f;

        private ICaravanAssetDurabilityReader reader;
        private ICaravanAssetDurabilityWriter writer;
        private IOwnedCaravanAssetReader owned;
        private bool isHubLoaded;
        private bool expanded;
        private Vector2 scroll;

        private void Awake()
        {
            reader = GetComponent<ICaravanAssetDurabilityReader>();
            writer = GetComponent<ICaravanAssetDurabilityWriter>();
        }

        private void Start()
        {
            var dependencies = FindFirstObjectByType<DependencyManager>(FindObjectsInactive.Include);
            if (dependencies != null) dependencies.TryResolve(out owned);
        }

        private void OnEnable()
        {
            isHubLoaded = SceneManager.GetSceneByName(SceneNames.Hub).isLoaded;
            SceneManager.sceneLoaded += HandleSceneLoaded;
            SceneManager.sceneUnloaded += HandleSceneUnloaded;
        }

        private void OnDisable()
        {
            SceneManager.sceneLoaded -= HandleSceneLoaded;
            SceneManager.sceneUnloaded -= HandleSceneUnloaded;
        }

        private void HandleSceneLoaded(Scene scene, LoadSceneMode mode)
        {
            if (scene.name == SceneNames.Hub) isHubLoaded = true;
        }

        private void HandleSceneUnloaded(Scene scene)
        {
            if (scene.name == SceneNames.Hub) isHubLoaded = false;
        }

        private void OnGUI()
        {
            if (!isHubLoaded) return;

            var toggle = new Rect(Margin, Screen.height - ToggleHeight - Margin, ToggleWidth, ToggleHeight);
            if (GUI.Button(toggle, expanded ? "내구도 닫기" : "내구도")) expanded = !expanded;
            if (!expanded) return;

            var area = new Rect(Margin, toggle.y - PanelHeight - 4f, PanelWidth, PanelHeight);
            GUI.Box(area, GUIContent.none);
            GUILayout.BeginArea(new Rect(area.x + 6f, area.y + 6f, area.width - 12f, area.height - 12f));
            if (reader == null || writer == null || owned == null)
            {
                GUILayout.Label("내구도 매니저·보유 목록 없음(Build Bootstrap Scene)");
                GUILayout.EndArea();
                return;
            }

            scroll = GUILayout.BeginScrollView(scroll);
            foreach (var kind in Kinds)
            {
                foreach (var id in owned.GetOwnedIds(kind))
                {
                    if (!owned.TryGetOwned(id, out var asset) || !reader.TryGet(id, out var current, out var max)) continue;
                    GUILayout.BeginHorizontal();
                    GUILayout.Label($"{OwnedCaravanAssetNames.Format(asset)} {CaravanAssetDurabilityRules.Display(current)}/{max}", GUILayout.Width(200f));
                    if (GUILayout.Button("-10%")) writer.TrySet(id, current - max * StepRatio);
                    if (GUILayout.Button("+10%")) writer.TrySet(id, current + max * StepRatio);
                    if (GUILayout.Button("1%")) writer.TrySet(id, max * NearZeroRatio);
                    GUILayout.EndHorizontal();
                }
            }
            GUILayout.EndScrollView();
            GUILayout.EndArea();
        }
    }
```
