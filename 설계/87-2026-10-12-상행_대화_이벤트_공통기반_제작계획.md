## Global Constraints

- 작업 브랜치 `system`. **커밋은 사용자 지시 시에만** — 이 계획의 태스크에는 커밋 단계가 없다. `Docs/`는 gitignore 대상(커밋하지 않음).
- `.unity`/`.prefab` 직접 편집 금지 — 씬 변경은 `Tools/Game/...` 인스톨러(get-or-create, 재실행 안전).
- 전역 매니저는 `IManagedComponent` + `ManagerHierarchyInstaller` 목록 등록. 매니저 하위 컴포넌트는 같은 GameObject, 매니저가 `GetComponent`로 조회(없으면 `InvalidOperationException`).
- DI는 등록한 **타입 그대로만** 조회된다 — 읽기·쓰기 계약을 각각 `Register`한다.
- 미설계 의존성은 `TryResolve`, 소비자는 null 처리.
- 주석은 한국어, "왜"를 우선. 명명: 인터페이스 `I`, 표시 전용 `XxxView`, UI Id 모음 `XxxUIElementIds`.

- 중요도 순서: `Scenario` > `Core` > `Side` > `Normal`. 같은 중요도 FIFO, 같은 판정 회차는 테이블 순서.
- 발생 기록은 **끝까지 진행했을 때만**. 중단 = 즉시 종료(무응답 분기 미적용, 기록 없음).

## 설계 보완 (계획 작성 중 확정 — 설계 87에 반영할 것)

- **질문 줄 태그 `#ask`**: Yarn은 줄 표시가 끝나야 다음 선택지를 넘겨주므로, 표시기는 "다음이 선택지인지" 미리 알 수 없다. 선택지 바로 앞 줄(대화)에 `#ask`를 붙이면 그 줄은 타자기 출력이 끝나는 즉시 완료되어(대기 없음) 선택지가 곧바로 뜬다. 이벤트는 `#choice` 줄이 같은 역할(89). 콘텐츠 검증이 "선택지 묶음 앞 줄에 `#ask` 또는 `#choice`"를 검사한다.
- **선택지가 하나도 고를 수 없을 때**: 보이는 선택지가 모두 잠겼고 `#timeout`도 없으면 선택 없음으로 넘긴다(Yarn 기본 동작: 선택지 묶음 건너뜀).

```csharp
using System;

namespace Game.Core
{
    /// <summary>
    /// 발생 조건 1개(설계 87 §5.1). 엑셀 조건 열을 임포트할 때 한 번 해석해 자산에 직렬화한다 - 실행 중 문자열 파싱을 반복하지 않기 위해서다.
    /// </summary>
    [Serializable]
    public struct EncounterCondition
    {
        public string Key;
        public bool Negated;
        public string Argument;

        public EncounterCondition(string key, bool negated, string argument)
        {
            Key = key;
            Negated = negated;
            Argument = argument;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    /// <summary>
    /// 조건 열 문자열(`region:2; !flag:x`)을 조건 목록으로 바꾼다(설계 87 §2.3). 인자 형식은 키별 판정기가 검사한다(EncounterConditionRegistry.IsValidArgument) -
    /// 여기서는 구문과 키만 본다.
    /// </summary>
    public static class EncounterConditionParser
    {
        public static List<EncounterCondition> Parse(string text)
        {
            var result = new List<EncounterCondition>();
            if (string.IsNullOrWhiteSpace(text)) return result;

            foreach (var raw in text.Split(';'))
            {
                var part = raw.Trim();
                if (part.Length == 0) continue;

                var negated = part[0] == '!';
                if (negated) part = part.Substring(1).TrimStart();

                var colon = part.IndexOf(':');
                if (colon <= 0 || colon == part.Length - 1)
                {
                    throw new FormatException($"조건 '{raw.Trim()}'은(는) '키:값' 형식이어야 한다.");
                }

                var key = part.Substring(0, colon).Trim();
                var argument = part.Substring(colon + 1).Trim();
                if (!EncounterConditionKeys.IsKnown(key))
                {
                    throw new FormatException($"알 수 없는 조건 키 '{key}' - 알려진 키: {string.Join(", ", EncounterConditionKeys.All)}.");
                }
                if (argument.Length == 0)
                {
                    throw new FormatException($"조건 '{raw.Trim()}'의 값이 비어 있다.");
                }

                result.Add(new EncounterCondition(key, negated, argument));
            }
            return result;
        }
    }
}
```

```csharp
using System.Collections.Generic;

namespace Game.Core
{
    /// <summary>
    /// 이력의 순수 구현(설계 87 §4). 플레이 세션 동안만 유지 - 저장 기능은 범위 밖이다(기획 86 §2). MonoBehaviour 껍데기(EncounterHistoryStore)가

    public sealed class EncounterHistory : IEncounterHistoryReader, IEncounterHistoryWriter, IEncounterScriptVariableStore
    {
        private readonly HashSet<string> flags = new();
        private readonly HashSet<string> occurred = new();
        private readonly HashSet<string> tripOccurred = new();
        private readonly Dictionary<string, object> scriptValues = new();

        public IReadOnlyCollection<string> ActiveFlags => flags;
        public IReadOnlyCollection<string> OccurredIds => occurred;
        public IReadOnlyCollection<string> TripOccurredIds => tripOccurred;
        public IReadOnlyDictionary<string, object> ScriptValues => scriptValues;

        public bool HasFlag(string flag) => !string.IsNullOrEmpty(flag) && flags.Contains(flag);
        public bool HasOccurred(string contentId) => !string.IsNullOrEmpty(contentId) && occurred.Contains(contentId);
        public bool OccurredThisTrip(string contentId) => !string.IsNullOrEmpty(contentId) && tripOccurred.Contains(contentId);

        public void SetFlag(string flag, bool value)
        {
            if (string.IsNullOrEmpty(flag)) return;
            if (value) flags.Add(flag);
            else flags.Remove(flag);
        }

        public void RecordCompleted(string contentId)
        {
            if (string.IsNullOrEmpty(contentId)) return;
            occurred.Add(contentId);
            tripOccurred.Add(contentId);
        }

        public void ResetTrip() => tripOccurred.Clear();

        public void ResetAll()
        {
            flags.Clear();
            occurred.Clear();
            tripOccurred.Clear();
            scriptValues.Clear();
        }

        public bool TryGetScriptValue(string name, out object value)
        {
            if (string.IsNullOrEmpty(name))
            {
                value = null;
                return false;
            }
            return scriptValues.TryGetValue(name, out value);
        }

        public void SetScriptValue(string name, object value)
        {
            if (string.IsNullOrEmpty(name)) return;
            scriptValues[name] = value;
        }
    }
}
```

```csharp
using System.Globalization;

namespace Game.Core
{
    /// <summary>조건 인자의 범위 표기 해석(설계 87 §5.1). `3` = 3~3, `3-5` = 3~5. 형식이 틀리면 false - 실행 중엔 예외 대신 "거짓"으로 처리한다.</summary>
    public static class EncounterRange
    {
        public static bool TryParseInt(string text, out int min, out int max)
        {
            min = max = 0;
            if (string.IsNullOrWhiteSpace(text)) return false;
            var parts = text.Split('-');
            if (parts.Length == 1)
            {
                if (!int.TryParse(parts[0].Trim(), NumberStyles.Integer, CultureInfo.InvariantCulture, out min)) return false;
                max = min;
                return true;
            }
            return parts.Length == 2
                && int.TryParse(parts[0].Trim(), NumberStyles.Integer, CultureInfo.InvariantCulture, out min)
                && int.TryParse(parts[1].Trim(), NumberStyles.Integer, CultureInfo.InvariantCulture, out max);
        }

        public static bool TryParseFloat(string text, out float min, out float max)
        {
            min = max = 0f;
            if (string.IsNullOrWhiteSpace(text)) return false;
            var parts = text.Split('-');
            if (parts.Length == 1)
            {
                if (!float.TryParse(parts[0].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out min)) return false;
                max = min;
                return true;
            }
            return parts.Length == 2
                && float.TryParse(parts[0].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out min)
                && float.TryParse(parts[1].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out max);
        }

        /// <summary>`월/일-월/일` → 해 안의 일 번호(0부터). 시작 > 끝이면 해를 넘는 범위다.</summary>
        public static bool TryParseDayOfYearRange(string text, out int start, out int end)
        {
            start = end = 0;
            if (string.IsNullOrWhiteSpace(text)) return false;
            var parts = text.Split('-');
            return parts.Length == 2 && TryParseMonthDay(parts[0], out start) && TryParseMonthDay(parts[1], out end);
        }

        /// <summary>순환 범위 포함 판정 - min ≤ max면 일반 범위, 아니면 끝을 넘어 처음으로 이어지는 범위.</summary>
        public static bool InCyclicRange(int value, int min, int max) => min <= max ? value >= min && value <= max : value >= min || value <= max;

        private static bool TryParseMonthDay(string text, out int dayOfYear)
        {
            dayOfYear = 0;
            var md = text.Trim().Split('/');
            if (md.Length != 2) return false;
            if (!int.TryParse(md[0], NumberStyles.Integer, CultureInfo.InvariantCulture, out var month)) return false;
            if (!int.TryParse(md[1], NumberStyles.Integer, CultureInfo.InvariantCulture, out var day)) return false;
            if (month < 1 || month > GameDate.MonthsPerYear || day < 1 || day > GameDate.DaysPerMonth) return false;
            dayOfYear = (month - 1) * GameDate.DaysPerMonth + (day - 1);
            return true;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Globalization;

namespace Game.Core
{
    /// <summary>조건 키 하나의 판정기. 새 조건 종류는 키(EncounterConditionKeys)와 이 구현 하나를 추가한다(기획 86 §5.3).</summary>
    public interface IEncounterConditionEvaluator
    {
        string Key { get; }
        bool IsValidArgument(string argument);
        bool Evaluate(string argument, EncounterConditionContext context);
    }

    /// <summary>
    /// 키별 판정기 등록부(설계 87 §5.1). 임포터는 IsValidArgument로 인자 형식을 미리 막고, 실행 중엔 형식이 틀리거나 키를 모르면 예외 없이 거짓이다 -
    /// 손으로 고친 자산 하나가 상행 전체를 멈추지 않게 하기 위해서다.
    /// </summary>
    public sealed class EncounterConditionRegistry
    {
        private readonly Dictionary<string, IEncounterConditionEvaluator> evaluators = new();

        public static EncounterConditionRegistry CreateDefault()
        {
            var registry = new EncounterConditionRegistry();
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.Month,
                a => EncounterRange.TryParseInt(a, out var lo, out var hi) && lo >= 1 && hi <= GameDate.MonthsPerYear && hi >= 1 && lo <= GameDate.MonthsPerYear,
                (a, c) => EncounterRange.TryParseInt(a, out var lo, out var hi) && EncounterRange.InCyclicRange(c.Today.Month, lo, hi)));
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.Date,
                a => EncounterRange.TryParseDayOfYearRange(a, out _, out _),
                (a, c) => EncounterRange.TryParseDayOfYearRange(a, out var s, out var e)
                    && EncounterRange.InCyclicRange((c.Today.Month - 1) * GameDate.DaysPerMonth + (c.Today.Day - 1), s, e)));
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.Progress,
                a => EncounterRange.TryParseFloat(a, out var lo, out var hi) && lo <= hi,
                (a, c) => EncounterRange.TryParseFloat(a, out var lo, out var hi) && c.Progress >= lo && c.Progress <= hi));
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.Region, IsInt,
                (a, c) => c.HasLeg && TryInt(a, out var r) && (c.FromRegionId == r || c.ToRegionId == r)));
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.Road,
                a => TryRoad(a, out _, out _),
                (a, c) => c.HasLeg && TryRoad(a, out var x, out var y)
                    && ((c.FromCityId == x && c.ToCityId == y) || (c.FromCityId == y && c.ToCityId == x))));
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.From, IsInt, (a, c) => c.HasLeg && TryInt(a, out var id) && c.FromCityId == id));
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.To, IsInt, (a, c) => c.HasLeg && TryInt(a, out var id) && c.ToCityId == id));
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.Flag, IsNonEmpty, (a, c) => c.History != null && c.History.HasFlag(a)));
            registry.Register(new DelegateEvaluator(EncounterConditionKeys.Seen, IsNonEmpty, (a, c) => c.History != null && c.History.HasOccurred(a)));
            return registry;
        }

        public void Register(IEncounterConditionEvaluator evaluator) => evaluators[evaluator.Key] = evaluator;

        public bool IsValidArgument(string key, string argument)
            => key != null && evaluators.TryGetValue(key, out var e) && e.IsValidArgument(argument);

        public bool Evaluate(EncounterCondition condition, EncounterConditionContext context)
        {
            if (condition.Key == null || !evaluators.TryGetValue(condition.Key, out var evaluator)) return false;
            if (!evaluator.IsValidArgument(condition.Argument)) return false;
            var result = evaluator.Evaluate(condition.Argument, context);
            return condition.Negated ? !result : result;
        }

        public bool EvaluateAll(IReadOnlyList<EncounterCondition> conditions, EncounterConditionContext context)
        {
            if (conditions == null) return true;
            for (var i = 0; i < conditions.Count; i++)
            {
                if (!Evaluate(conditions[i], context)) return false;
            }
            return true;
        }

        private static bool IsNonEmpty(string a) => !string.IsNullOrWhiteSpace(a);
        private static bool IsInt(string a) => TryInt(a, out _);
        private static bool TryInt(string a, out int value) => int.TryParse(a?.Trim(), NumberStyles.Integer, CultureInfo.InvariantCulture, out value);

        // `a-b` = 두 도시를 잇는 도로(방향 무관).
        private static bool TryRoad(string a, out int x, out int y)
        {
            x = y = 0;
            var parts = a?.Split('-');
            return parts != null && parts.Length == 2 && TryInt(parts[0], out x) && TryInt(parts[1], out y);
        }

        private sealed class DelegateEvaluator : IEncounterConditionEvaluator
        {
            private readonly Func<string, bool> isValid;
            private readonly Func<string, EncounterConditionContext, bool> evaluate;

            public DelegateEvaluator(string key, Func<string, bool> isValid, Func<string, EncounterConditionContext, bool> evaluate)
            {
                Key = key;
                this.isValid = isValid;
                this.evaluate = evaluate;
            }

            public string Key { get; }
            public bool IsValidArgument(string argument) => isValid(argument);
            public bool Evaluate(string argument, EncounterConditionContext context) => evaluate(argument, context);
        }
```

```csharp
namespace Game.Core
```

```csharp
    /// <summary>
    /// 대화·이벤트 공용 후보 거르기와 가중치 추첨(설계 87 §5.2). 순수 함수 묶음 - 난수는 주입받아 테스트에서 고정한다.
    /// 횟수 제한: 이벤트 전부(Once = 게임 전체, PerTrip = 이번 상행), 대화는 랜덤·수동만 이번 상행 1회. 계기(확정) 대화와 반복 대화는 제한 없다
    /// (기획 86 §4.3, §8-9 - 확정 대화 제한은 실전 테스트 후 결정).
    /// </summary>
    public static class EncounterCandidatePicker
    {
        public static bool IsEventExhausted(EventEntry entry, IEncounterHistoryReader history)
            => entry.Repeat == EventRepeat.Once ? history.HasOccurred(entry.Id) : history.OccurredThisTrip(entry.Id);

        public static bool IsDialogueExhausted(DialogueEntry entry, IEncounterHistoryReader history)
            => (entry.Occurrence == DialogueOccurrence.Random || entry.Occurrence == DialogueOccurrence.Manual) && history.OccurredThisTrip(entry.Id);

        public static List<EventEntry> FilterEvents(IReadOnlyList<EventEntry> entries, EventOccurrence occurrence, IEncounterHistoryReader history,
            EncounterConditionRegistry conditions, EncounterConditionContext context)
        {
            var result = new List<EventEntry>();
            foreach (var entry in entries)
            {
                if (entry.Occurrence != occurrence) continue;
                if (IsEventExhausted(entry, history)) continue;
                if (!conditions.EvaluateAll(entry.Conditions, context)) continue;
                result.Add(entry);
            }
            return result;
        }

        public static List<DialogueEntry> FilterDialogues(IReadOnlyList<DialogueEntry> entries, Func<DialogueEntry, bool> kind, IEncounterHistoryReader history,
            Func<string, bool> isSpeakerAvailable, EncounterConditionRegistry conditions, EncounterConditionContext context)
        {
            var result = new List<DialogueEntry>();
            foreach (var entry in entries)
            {
                if (!kind(entry)) continue;
                if (IsDialogueExhausted(entry, history)) continue;
                if (!AllSpeakersAvailable(entry, isSpeakerAvailable)) continue;
                if (!conditions.EvaluateAll(entry.Conditions, context)) continue;
                result.Add(entry);
            }
            return result;
        }

        public static T PickWeighted<T>(IReadOnlyList<T> candidates, Func<T, int> weightOf, Func<float> random01) where T : class
        {
            if (candidates == null || candidates.Count == 0) return null;

            var total = 0;
            foreach (var c in candidates) total += Math.Max(0, weightOf(c));
            if (total <= 0) return candidates[0];

            var roll = random01() * total;
            foreach (var c in candidates)
            {
                roll -= Math.Max(0, weightOf(c));
                if (roll < 0f) return c;
            }
            return candidates[candidates.Count - 1];
        }

        private static bool AllSpeakersAvailable(DialogueEntry entry, Func<string, bool> isSpeakerAvailable)
        {
            if (entry.Speakers == null) return true;
            foreach (var speaker in entry.Speakers)
            {
                if (!isSpeakerAvailable(speaker)) return false;
            }
            return true;
        }
    }
}
```

```csharp
    /// ③ 같은 회차 → 테이블 순서. 꺼낼 때 조건을 다시 검사해 깨진 것은 버린다 - 버려진 이벤트가 다시 성립하면 새 회차로 들어가 그 중요도의 맨 뒤에 선다.
    /// 판정은 5초 간격이라 같은 회차 안의 성립 순서는 구분할 수 없어 테이블 순서를 보조 기준으로 쓴다.
    /// </summary>
    public sealed class FixedEventQueue
    {
        private sealed class Item
        {
            public EventEntry Entry;
            public int TableIndex;
            public long Round;
        }

        private readonly List<Item> items = new();
        private long round;

        public int Count => items.Count;

        public IReadOnlyList<EventEntry> Snapshot()
        {
            var list = new List<EventEntry>(items.Count);
            foreach (var item in items) list.Add(item.Entry);
            return list;
        }

        public bool Contains(string id)
        {
            foreach (var item in items)
            {
                if (item.Entry.Id == id) return true;
            }
            return false;
        }

        /// <summary>이번 판정 회차에 성립한 확정 이벤트(테이블 순서)를 넣는다. 이미 대기 중인 것은 건너뛴다.</summary>
        public void EnqueueSatisfied(IReadOnlyList<(EventEntry entry, int tableIndex)> satisfiedInTableOrder)
        {
            round++;
            var added = false;
            foreach (var (entry, tableIndex) in satisfiedInTableOrder)
            {
                if (entry == null || Contains(entry.Id)) continue;
                items.Add(new Item { Entry = entry, TableIndex = tableIndex, Round = round });
                added = true;
            }
            if (added) items.Sort(Compare);
        }

        public bool TryDequeue(Func<EventEntry, bool> stillValid, out EventEntry entry)
        {
            while (items.Count > 0)
            {
                var head = items[0];
                items.RemoveAt(0);
                if (stillValid(head.Entry))
                {
                    entry = head.Entry;
                    return true;
                }
            }
            entry = null;
            return false;
        }

        public void Clear() => items.Clear();

        // 세 키가 항목마다 유일한 순서를 만들므로 List.Sort의 불안정성은 결과에 영향이 없다.
        private static int Compare(Item a, Item b)
        {
            var byImportance = a.Entry.Importance.CompareTo(b.Entry.Importance);
            if (byImportance != 0) return byImportance;
            var byRound = a.Round.CompareTo(b.Round);
            return byRound != 0 ? byRound : a.TableIndex.CompareTo(b.TableIndex);
        }
    }
}
```

```csharp
using System.Collections.Generic;

namespace Game.Core
{
    public enum TripHaltTransition { None, Started, StillHeld, Ended }

    /// <summary>정지 원인 집합(설계 87 §6.2). 전이만 계산하고 부수 효과는 TripHaltController가 적용한다.</summary>
    public sealed class TripHaltState
    {
        private readonly HashSet<TripHaltReason> reasons = new();

        public bool IsHalted => reasons.Count > 0;
        public bool IsHeld(TripHaltReason reason) => reasons.Contains(reason);

        public TripHaltTransition Acquire(TripHaltReason reason)
        {
            if (!reasons.Add(reason)) return TripHaltTransition.None;
            return reasons.Count == 1 ? TripHaltTransition.Started : TripHaltTransition.StillHeld;
        }

        public TripHaltTransition Release(TripHaltReason reason)
        {
            if (!reasons.Remove(reason)) return TripHaltTransition.None;
            return reasons.Count == 0 ? TripHaltTransition.Ended : TripHaltTransition.StillHeld;
        }

        public void Clear() => reasons.Clear();
    }
}
```

```csharp
using System;

namespace Game.Core
{
    /// <summary>
    /// 정지 원인 전이를 상행 시계·배치 활동에 적용한다(설계 87 §6.2). 원인이 하나라도 남으면 활동을 다시 전부 멈춘다 - 전투는 시작 시 배치 활동만
    /// 재개하므로(LiveBattleSimulationRule), 이벤트 전투가 끝나 "전투"만 풀렸을 때 그 재개를 되돌려야 이벤트 진행 중 배치가 흐르지 않는다.
    /// 순수 C#이라 EditMode에서 검증하고, MonoBehaviour 껍데기(TripHaltCoordinator)가 FieldManager 하위에서 들고 있다.
    /// </summary>
    public sealed class TripHaltController : ITripHaltControl
    {
        private readonly TripHaltState state = new();
        private readonly ISessionState session;
        private readonly IFieldFormationActivityRepository activities; // 선택적

        public TripHaltController(ISessionState session, IFieldFormationActivityRepository activities)
        {
            this.session = session;
            this.activities = activities;
        }

        public bool IsHalted => state.IsHalted;
        public bool IsHeld(TripHaltReason reason) => state.IsHeld(reason);
        public event Action HaltStarted;
        public event Action HaltEnded;

        public void Acquire(TripHaltReason reason)
        {
            switch (state.Acquire(reason))
            {
                case TripHaltTransition.Started:
                    session.Pause();
                    activities?.PauseAll();
                    HaltStarted?.Invoke();
                    break;
                case TripHaltTransition.StillHeld:
                    activities?.PauseAll();
                    break;
            }
        }

        public void Release(TripHaltReason reason)
        {
            switch (state.Release(reason))
            {
                case TripHaltTransition.Ended:
                    activities?.ResumeAll();
                    session.Resume();
                    HaltEnded?.Invoke();
                    break;
                case TripHaltTransition.StillHeld:
                    activities?.PauseAll();
                    break;
            }
        }

        public void ClearAll() => state.Clear();
    }
}
```

```csharp
using System.Collections.Generic;

namespace Game.Core
{
    public enum TripCheckKind { Battle, Dialogue, Event }

    /// <summary>
    /// 판정 시점 계산(설계 87 §6.3). 전투 0초 지점(5초마다), 대화 +1.25초(20초마다), 이벤트 +2.5초(5초마다) - 세 판정이 같은 순간에 겹치지 않게
    /// 한 시간 축에서 어긋나게 둔다(기획 86 §3.2). 한 번에 여러 경계를 넘으면 시각 순서대로 모두 낸다.
    /// </summary>
    public sealed class TripCheckSchedule
    {
        public const float BattleIntervalSeconds = 5f;
        public const float DialogueOffsetSeconds = 1.25f;
        public const float DialogueIntervalSeconds = 20f;
        public const float EventOffsetSeconds = 2.5f;
        public const float EventIntervalSeconds = 5f;

        private float nextBattle;
        private float nextDialogue;
        private float nextEvent;

        public float Elapsed { get; private set; }

        public TripCheckSchedule() => Reset();

        public void Reset()
        {
            Elapsed = 0f;
            nextBattle = BattleIntervalSeconds;
            nextDialogue = DialogueOffsetSeconds;
            nextEvent = EventOffsetSeconds;
        }

        public float SecondsUntil(TripCheckKind kind) => NextOf(kind) - Elapsed;

        public void Advance(float delta, List<TripCheckKind> fired)
        {
            fired.Clear();
            if (delta <= 0f) return;
            Elapsed += delta;

            while (true)
            {
                var kind = Earliest();
                if (NextOf(kind) > Elapsed + 1e-5f) break;
                fired.Add(kind);
                switch (kind)
                {
                    case TripCheckKind.Battle: nextBattle += BattleIntervalSeconds; break;
                    case TripCheckKind.Dialogue: nextDialogue += DialogueIntervalSeconds; break;
                    default: nextEvent += EventIntervalSeconds; break;
                }
            }
        }

        private float NextOf(TripCheckKind kind) => kind switch
        {
            TripCheckKind.Battle => nextBattle,
            TripCheckKind.Dialogue => nextDialogue,
            _ => nextEvent,
        };

        // 오프셋 설계상 같은 시각은 나오지 않지만, 같으면 전투 → 대화 → 이벤트 순.
        private TripCheckKind Earliest()
        {
            var kind = TripCheckKind.Battle;
            if (nextDialogue < NextOf(kind)) kind = TripCheckKind.Dialogue;
            if (nextEvent < NextOf(kind)) kind = TripCheckKind.Event;
            return kind;
        }
    }
}
```

```csharp
    /// <summary>판정 신호 수신(설계 87 §6.3).</summary>
    public interface ITripCheckSignal
    {
        event Action BattleCheck;
        event Action DialogueCheck;
        event Action EventCheck;
        float SecondsUntil(TripCheckKind kind);
    }

    /// <summary>
    /// FieldManager 하위 컴포넌트 - 상행 진행 이벤트(OnProgressChanged)에 얹어 시간을 센다. 진행 이벤트는 정지 중엔 오지 않으므로 전투·이벤트 중 판정이
    /// 별도 코드 없이 멈춘다. 도착 틱(진행도 ≥ 1)은 세지 않는다 - 예전 EncounterManager의 "도착과 인카운터 동시 처리" 버그 방지를 그대로 옮겼다.
    /// </summary>
    public class TripCheckClock : MonoBehaviour, ITripCheckSignal
    {
        private readonly TripCheckSchedule schedule = new();
        private readonly List<TripCheckKind> fired = new();
        private ISessionState session;
        private ITripMilestoneSource milestones;

        public event Action BattleCheck;
        public event Action DialogueCheck;
        public event Action EventCheck;

        public float SecondsUntil(TripCheckKind kind) => schedule.SecondsUntil(kind);

        public void Bind(ISessionState session, ITripMilestoneSource milestones)
        {
            Unbind();
            this.session = session;
            this.milestones = milestones;
            if (session != null) session.OnProgressChanged += HandleProgressChanged;
            if (milestones != null) milestones.Raised += HandleMilestone;
        }

        private void HandleProgressChanged(float progress)
        {
            if (progress >= 1f) return;
            schedule.Advance(Time.deltaTime, fired);
            foreach (var kind in fired)
            {
                switch (kind)
                {
                    case TripCheckKind.Battle: BattleCheck?.Invoke(); break;
                    case TripCheckKind.Dialogue: DialogueCheck?.Invoke(); break;
                    default: EventCheck?.Invoke(); break;
                }
            }
        }

        private void HandleMilestone(TripMilestone milestone)
        {
            if (milestone == TripMilestone.TripStarted) schedule.Reset();
        }

        private void Unbind()
        {
            if (session != null) session.OnProgressChanged -= HandleProgressChanged;
            if (milestones != null) milestones.Raised -= HandleMilestone;
        }

        private void OnDestroy() => Unbind();
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using UnityEditor;
using UnityEngine;

namespace Game.Core.Editor
{
    /// <summary>
    /// Assets/Table/Encounter/Encounter.xlsx → EncounterTableAsset(Docs/설계/87번 §3.1). 조건은 임포트 때 한 번 해석해 자산에 넣는다 - 키·인자 오류를
    /// 실행 전에 막기 위해서다. 화자는 캐릭터 원본 엑셀(CharacterStats)과 Person 시트로 검사한다(임포트 순서와 무관하게 같은 결과). 열거형 열은 이름
    /// 문자열이다(행을 읽는 사람이 바로 알아보도록, TownStock과 같은 방식). YarnNode 존재는 Yarn 컴파일 결과가 필요해 콘텐츠 검증 테스트가 본다(§9).
    /// </summary>
    public static class EncounterTableImporter
    {
        private const int DefaultWeight = 100;

        [MenuItem("Tools/Game/Table/Import Encounter")]
        public static void Import()
        {
            var workbookPath = TablePath(EncounterTableTemplateWriter.RelativePath);
            if (!File.Exists(workbookPath))
            {
                Debug.LogError($"{nameof(EncounterTableImporter)}: 워크북을 찾을 수 없다 - '{workbookPath}' (Tools > Game > Table > Create Encounter Template).");
                return;
            }

            try
            {
                var registry = EncounterConditionRegistry.CreateDefault();
                var people = ReadPeople(workbookPath);
                var events = ReadEvents(workbookPath, registry);
                var dialogues = ReadDialogues(workbookPath, registry);
                Validate(events, dialogues, people);

                var asset = EditorTableReader.GetOrCreateAsset<EncounterTableAsset>(TableAssetPaths.EncounterTable);
                Write(asset, events, dialogues, people);
                AssetDatabase.SaveAssets();
                Debug.Log($"{nameof(EncounterTableImporter)}: 임포트 완료 - 이벤트 {events.Count}, 대화 {dialogues.Count}, 외부 인물 {people.Count}.");
            }
            catch (FormatException e)
            {
                Debug.LogError($"{nameof(EncounterTableImporter)}: {e.Message}");
            }
        }

        private static string TablePath(string relative) => Path.Combine(Application.dataPath, "Table", relative);

        private static List<EventEntry> ReadEvents(string path, EncounterConditionRegistry registry)
        {
            var list = new List<EventEntry>();
            foreach (var row in EditorTableReader.ReadSheet(path, "Event"))
            {
                var id = EditorTableReader.ParseRequiredString(row, "Id").Trim();
                list.Add(new EventEntry
                {
                    Id = id,
                    YarnNode = EditorTableReader.ParseRequiredString(row, "YarnNode").Trim(),
                    Occurrence = ParseName<EventOccurrence>(row, "Occurrence", id),
                    Importance = ParseName<EventImportance>(row, "Importance", id),
                    Repeat = ParseName<EventRepeat>(row, "Repeat", id),
                    Weight = ParseWeight(row, id),
                    Conditions = ParseConditions(row, id, registry),
                });
            }
            return list;
        }

        private static List<DialogueEntry> ReadDialogues(string path, EncounterConditionRegistry registry)
        {
            var list = new List<DialogueEntry>();
            foreach (var row in EditorTableReader.ReadSheet(path, "Dialogue"))
            {
                var id = EditorTableReader.ParseRequiredString(row, "Id").Trim();
                var occurrence = ParseName<DialogueOccurrence>(row, "Occurrence", id);
                var trigger = Optional(row, "Trigger") is { } t ? ParseNameText<DialogueTrigger>(t, "Trigger", id) : DialogueTrigger.None;
                if (occurrence == DialogueOccurrence.Trigger && trigger == DialogueTrigger.None) throw new FormatException($"Dialogue '{id}': Occurrence가 Trigger면 Trigger 열이 필요하다.");
                if (occurrence != DialogueOccurrence.Trigger && trigger != DialogueTrigger.None) throw new FormatException($"Dialogue '{id}': Trigger 열은 Occurrence가 Trigger일 때만 쓴다.");

                list.Add(new DialogueEntry
                {
                    Id = id,
                    YarnNode = EditorTableReader.ParseRequiredString(row, "YarnNode").Trim(),
                    Occurrence = occurrence,
                    Trigger = trigger,
                    Speakers = ParseSpeakers(row),
                    Weight = ParseWeight(row, id),
                    Conditions = ParseConditions(row, id, registry),
                });
            }
            return list;
        }

        private static List<PersonEntry> ReadPeople(string path)
        {
            var names = new Dictionary<string, string>();
            foreach (var row in EditorTableReader.ReadSheet(path, "PersonStrings"))
            {
                names[EditorTableReader.ParseRequiredString(row, "Id").Trim()] = EditorTableReader.ParseRequiredString(row, "Name");
            }

            var list = new List<PersonEntry>();
            foreach (var row in EditorTableReader.ReadSheet(path, "Person"))
            {
                var id = EditorTableReader.ParseRequiredString(row, "Id").Trim();
                if (!names.TryGetValue(id, out var name)) throw new FormatException($"Person '{id}'의 이름이 PersonStrings 시트에 없다.");
                list.Add(new PersonEntry { Id = id, Name = name, IllustrationKey = Optional(row, "IllustrationKey") ?? string.Empty });
            }
            return list;
        }

        private static void Validate(List<EventEntry> events, List<DialogueEntry> dialogues, List<PersonEntry> people)
        {
            var ids = new HashSet<string>();
            foreach (var e in events) if (!ids.Add(e.Id)) throw new FormatException($"Id '{e.Id}' 중복(이벤트·대화 Id는 이력을 함께 쓰므로 서로도 겹치면 안 된다).");
            foreach (var d in dialogues) if (!ids.Add(d.Id)) throw new FormatException($"Id '{d.Id}' 중복(이벤트·대화 Id는 이력을 함께 쓰므로 서로도 겹치면 안 된다).");

            var personIds = new HashSet<string>();
            foreach (var p in people) if (!personIds.Add(p.Id)) throw new FormatException($"Person Id '{p.Id}' 중복.");

            var characterIds = ReadCharacterIds();
            foreach (var d in dialogues)
            {
                foreach (var speaker in d.Speakers)
                {
                    if (!characterIds.Contains(speaker) && !personIds.Contains(speaker))
                    {
                        throw new FormatException($"Dialogue '{d.Id}': 화자 '{speaker}'가 CharacterStats·Person 어디에도 없다.");
                    }
                }
            }
        }

        private static HashSet<string> ReadCharacterIds()
        {
            var ids = new HashSet<string>();
            foreach (var row in EditorTableReader.ReadSheet(TablePath("Character/CharacterStats.xlsx"), "CharacterStats"))
            {
                if (row.TryGetValue("Id", out var id) && !string.IsNullOrWhiteSpace(id)) ids.Add(id.Trim());
            }
            return ids;
        }

        private static List<EncounterCondition> ParseConditions(IReadOnlyDictionary<string, string> row, string id, EncounterConditionRegistry registry)
        {
            List<EncounterCondition> conditions;
            try { conditions = EncounterConditionParser.Parse(Optional(row, "Conditions")); }
            catch (FormatException e) { throw new FormatException($"'{id}' 조건: {e.Message}"); }

            foreach (var c in conditions)
            {
                if (!registry.IsValidArgument(c.Key, c.Argument)) throw new FormatException($"'{id}' 조건 '{c.Key}:{c.Argument}'의 값 형식이 틀렸다.");
            }
            return conditions;
        }

        private static List<string> ParseSpeakers(IReadOnlyDictionary<string, string> row)
        {
            var list = new List<string>();
            var text = Optional(row, "Speakers");
            if (text == null) return list;
            foreach (var part in text.Split(';'))
            {
                var s = part.Trim();
                if (s.Length > 0) list.Add(s);
            }
            return list;
        }

        private static int ParseWeight(IReadOnlyDictionary<string, string> row, string id)
        {
            var text = Optional(row, "Weight");
            if (text == null) return DefaultWeight;
            if (!int.TryParse(text, out var weight) || weight < 0) throw new FormatException($"'{id}' Weight '{text}'는 0 이상의 정수여야 한다.");
            return weight;
        }

        private static TEnum ParseName<TEnum>(IReadOnlyDictionary<string, string> row, string column, string id) where TEnum : struct, Enum
            => ParseNameText<TEnum>(EditorTableReader.ParseRequiredString(row, column), column, id);

        // 숫자 문자열("0")도 Enum.TryParse를 통과하므로 이름으로 정의된 값만 받는다(TownStockTableImporter와 같은 이유).
        private static TEnum ParseNameText<TEnum>(string raw, string column, string id) where TEnum : struct, Enum
        {
            var text = raw.Trim();
            if (text.Length > 0 && !char.IsDigit(text[0]) && Enum.TryParse(text, ignoreCase: false, out TEnum value) && Enum.IsDefined(typeof(TEnum), value)) return value;
            throw new FormatException($"'{id}' {column} '{text}'은(는) {string.Join("/", Enum.GetNames(typeof(TEnum)))} 중 하나여야 한다.");
        }

        private static string Optional(IReadOnlyDictionary<string, string> row, string column)
            => row.TryGetValue(column, out var value) && !string.IsNullOrWhiteSpace(value) ? value.Trim() : null;

        private static void Write(EncounterTableAsset asset, List<EventEntry> events, List<DialogueEntry> dialogues, List<PersonEntry> people)
        {
            // 자산 필드가 private이라 SerializedObject로 쓴다.
            var so = new SerializedObject(asset);
            WriteEvents(so.FindProperty("events"), events);
            WriteDialogues(so.FindProperty("dialogues"), dialogues);
            WritePeople(so.FindProperty("people"), people);
            so.ApplyModifiedProperties();
            EditorUtility.SetDirty(asset);
        }

        private static void WriteEvents(SerializedProperty list, List<EventEntry> events)
        {
            list.arraySize = events.Count;
            for (var i = 0; i < events.Count; i++)
            {
                var e = list.GetArrayElementAtIndex(i);
                e.FindPropertyRelative("Id").stringValue = events[i].Id;
                e.FindPropertyRelative("YarnNode").stringValue = events[i].YarnNode;
                EditorTableReader.SetEnumValue(e.FindPropertyRelative("Occurrence"), events[i].Occurrence);
                EditorTableReader.SetEnumValue(e.FindPropertyRelative("Importance"), events[i].Importance);
                EditorTableReader.SetEnumValue(e.FindPropertyRelative("Repeat"), events[i].Repeat);
                e.FindPropertyRelative("Weight").intValue = events[i].Weight;
                WriteConditions(e.FindPropertyRelative("Conditions"), events[i].Conditions);
            }
        }

        private static void WriteDialogues(SerializedProperty list, List<DialogueEntry> dialogues)
        {
            list.arraySize = dialogues.Count;
            for (var i = 0; i < dialogues.Count; i++)
            {
                var d = list.GetArrayElementAtIndex(i);
                d.FindPropertyRelative("Id").stringValue = dialogues[i].Id;
                d.FindPropertyRelative("YarnNode").stringValue = dialogues[i].YarnNode;
                EditorTableReader.SetEnumValue(d.FindPropertyRelative("Occurrence"), dialogues[i].Occurrence);
                EditorTableReader.SetEnumValue(d.FindPropertyRelative("Trigger"), dialogues[i].Trigger);
                var speakers = d.FindPropertyRelative("Speakers");
                speakers.arraySize = dialogues[i].Speakers.Count;
                for (var s = 0; s < dialogues[i].Speakers.Count; s++) speakers.GetArrayElementAtIndex(s).stringValue = dialogues[i].Speakers[s];
                d.FindPropertyRelative("Weight").intValue = dialogues[i].Weight;
                WriteConditions(d.FindPropertyRelative("Conditions"), dialogues[i].Conditions);
            }
        }

        private static void WritePeople(SerializedProperty list, List<PersonEntry> people)
        {
            list.arraySize = people.Count;
            for (var i = 0; i < people.Count; i++)
            {
                var p = list.GetArrayElementAtIndex(i);
                p.FindPropertyRelative("Id").stringValue = people[i].Id;
                p.FindPropertyRelative("Name").stringValue = people[i].Name;
                p.FindPropertyRelative("IllustrationKey").stringValue = people[i].IllustrationKey;
            }
        }

        private static void WriteConditions(SerializedProperty list, List<EncounterCondition> conditions)
        {
            list.arraySize = conditions.Count;
            for (var i = 0; i < conditions.Count; i++)
            {
                var c = list.GetArrayElementAtIndex(i);
                c.FindPropertyRelative("Key").stringValue = conditions[i].Key;
                c.FindPropertyRelative("Negated").boolValue = conditions[i].Negated;
                c.FindPropertyRelative("Argument").stringValue = conditions[i].Argument;
            }
        }
    }
```

```csharp
namespace Game.Core
{
    /// <summary>
    /// 글자 수 기준은 공백 제외·문장부호 포함(기획 86 §4.5). 공백은 시간을 쓰지 않으므로, k번째 글자가 찍히는 순간 뒤따르는 공백까지 함께 보인다.
    /// </summary>
    public static class DialogueTextMetrics
    {
        public static int CountTyped(string text)
        {
            if (string.IsNullOrEmpty(text)) return 0;
            var count = 0;
            foreach (var c in text) if (!char.IsWhiteSpace(c)) count++;
            return count;
        }

        /// <summary>typedCount개 글자가 찍혔을 때 보이는 앞부분 길이(뒤따르는 공백 포함).</summary>
        public static int VisibleLengthFor(string text, int typedCount)
        {
            if (string.IsNullOrEmpty(text)) return 0;
            var seen = 0;
            for (var i = 0; i < text.Length; i++)
            {
                if (char.IsWhiteSpace(text[i])) continue;
                if (seen == typedCount) return i;
                seen++;
            }
            return text.Length;
        }
    }
```

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    public enum DialogueLinePhase { Idle, Typing, Waiting, Finished }

    /// <summary>
    /// 대사 한 줄의 진행 상태기(설계 87 §7). 타자기(초당 12자) → 대기(길이 비례) → 완료. 클릭은 2단계(출력 중 = 전체 표시, 출력 후 = 완료).
    /// 자동 넘김은 대화만 켠다(이벤트는 클릭으로만). finishOnTypingComplete = 선택지 앞 질문 줄(#ask) - 출력이 끝나면 대기 없이 완료해 선택지가 곧바로 뜬다.
    /// 일시정지 원인이 하나라도 있으면 출력·대기·클릭이 모두 멈춘다(모달 화면이 열린 동안).
    /// </summary>
    public sealed class DialogueLinePlayback
    {
        private readonly bool autoAdvance;
        private readonly HashSet<object> pauseReasons = new();
        private int typedTotal;
        private float typingElapsed;
        private float waitElapsed;
        private float waitDuration;
        private bool finishOnTypingComplete;

        public DialogueLinePlayback(bool autoAdvance) => this.autoAdvance = autoAdvance;

        public DialogueLinePhase Phase { get; private set; } = DialogueLinePhase.Idle;
        public string Text { get; private set; } = string.Empty;
        public int VisibleLength { get; private set; }
        public bool IsPaused => pauseReasons.Count > 0;

        public event Action TypingCompleted;
        public event Action Finished;

        public void Begin(string text, bool finishOnTypingComplete)
        {
            Text = text ?? string.Empty;
            this.finishOnTypingComplete = finishOnTypingComplete;
            typedTotal = DialogueTextMetrics.CountTyped(Text);
            typingElapsed = 0f;
            waitElapsed = 0f;
            waitDuration = DialoguePacing.WaitSecondsFor(typedTotal);
            Phase = DialogueLinePhase.Typing;
            VisibleLength = DialogueTextMetrics.VisibleLengthFor(Text, 0);
            if (typedTotal == 0) CompleteTyping();
        }

        public void Tick(float delta)
        {
            if (IsPaused || delta <= 0f) return;

            if (Phase == DialogueLinePhase.Typing)
            {
                typingElapsed += delta;
                var typed = Math.Min(typedTotal, (int)(typingElapsed * DialoguePacing.CharactersPerSecond + 1e-4f));
                VisibleLength = DialogueTextMetrics.VisibleLengthFor(Text, typed);
                if (typed >= typedTotal) CompleteTyping();
                return;
            }

            if (Phase == DialogueLinePhase.Waiting && autoAdvance)
            {
                waitElapsed += delta;
                if (waitElapsed >= waitDuration) Finish();
            }
        }

        public void Click()
        {
            if (IsPaused) return;
            if (Phase == DialogueLinePhase.Typing) CompleteTyping();
            else if (Phase == DialogueLinePhase.Waiting) Finish();
        }

        public void SetPaused(object reason, bool paused)
        {
            if (reason == null) return;
            if (paused) pauseReasons.Add(reason);
            else pauseReasons.Remove(reason);
        }

        /// <summary>중단 - 완료 이벤트 없이 멈춘다(설계 87 §7.2 즉시 종료).</summary>
        public void Stop()
        {
            Phase = DialogueLinePhase.Idle;
        }

        private void CompleteTyping()
        {
            VisibleLength = Text.Length;
            Phase = DialogueLinePhase.Waiting;
            waitElapsed = 0f;
            TypingCompleted?.Invoke();
            if (finishOnTypingComplete && Phase == DialogueLinePhase.Waiting) Finish();
        }

        private void Finish()
        {
            Phase = DialogueLinePhase.Finished;
            Finished?.Invoke();
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    /// <summary>재생 옵션 - 대화(자동 넘김, 선택지 10초)와 이벤트(클릭만, 제한 없음)의 차이만 담는다(설계 87 §7.2).</summary>
    public readonly struct DialoguePlaybackOptions
    {
        public bool AutoAdvance { get; }
        /// <summary>0 이하 = 제한 없음.</summary>
        public float OptionTimeoutSeconds { get; }

        public DialoguePlaybackOptions(bool autoAdvance, float optionTimeoutSeconds)
        {
            AutoAdvance = autoAdvance;
            OptionTimeoutSeconds = optionTimeoutSeconds;
        }

        public static DialoguePlaybackOptions Trip => new(true, DialoguePacing.DialogueOptionTimeoutSeconds);
        public static DialoguePlaybackOptions Event => new(false, 0f);
    }

    /// <summary>
    /// 줄 상태기 + 화면 + 선택지 제한시간을 묶는다(설계 87 §7.1). Yarn 연결 표시기가 PresentLine/PresentOptions를 부르고, 소유자(88·89 진행자)가 매 프레임
    /// Tick·클릭·일시정지를 전달한다. 무응답(#timeout) 선택지는 화면에 내보내지 않고 시간이 다 됐을 때만 고른다. 보이는 선택지가 모두 잠겼고 무응답도
    /// 없으면 선택 없음(-1)으로 넘긴다 - 대본이 멈추지 않게.
    /// </summary>
    public sealed class DialoguePlaybackController : IDialoguePresentationTarget
    {
        private readonly IDialogueView view;
        private readonly DialoguePlaybackOptions options;
        private readonly DialogueLinePlayback line;
        private readonly HashSet<object> pauseReasons = new();

        private Action onAdvance;
        private Action<int> onChosen;
        private int timeoutIndex = -1;
        private float optionElapsed;

        public DialoguePlaybackController(IDialogueView view, DialoguePlaybackOptions options)
        {
            this.view = view;
            this.options = options;
            line = new DialogueLinePlayback(options.AutoAdvance);
            line.TypingCompleted += () => view.SetTypingComplete(true);
            line.Finished += HandleLineFinished;
        }

        public bool IsShowingOptions => onChosen != null;

        public void PresentLine(DialogueLineData data, Action onAdvance)
        {
            this.onAdvance = onAdvance;
            view.HideOptions();
            view.ShowLine(data);
            view.SetTypingComplete(false);
            line.Begin(data.Text, finishOnTypingComplete: data.HasTag(DialogueTags.Ask));
            view.SetVisibleLength(line.VisibleLength);
        }

        public void PresentOptions(IReadOnlyList<DialogueOptionData> all, Action<int> onChosen)
        {
            var visible = new List<DialogueOptionData>();
            timeoutIndex = -1;
            var anyAvailable = false;
            foreach (var option in all)
            {
                if (option.IsTimeout)
                {
                    timeoutIndex = option.Index;
                    continue;
                }
                visible.Add(option);
                anyAvailable |= option.IsAvailable;
            }

            if (!anyAvailable && timeoutIndex < 0)
            {
                onChosen(-1);
                return;
            }

            this.onChosen = onChosen;
            optionElapsed = 0f;
            view.ShowOptions(visible, index => Choose(index, requireAvailable: true, visible));
            if (HasTimeout) view.SetOptionTimeRemaining(1f);
        }

        public void EndPresentation() => view.Hide();

        public void Tick(float delta)
        {
            line.Tick(delta);
            if (IsShowingOptions && HasTimeout && pauseReasons.Count == 0 && delta > 0f)
            {
                optionElapsed += delta;
                view.SetOptionTimeRemaining(Math.Max(0f, 1f - optionElapsed / options.OptionTimeoutSeconds));
                if (optionElapsed >= options.OptionTimeoutSeconds) Choose(timeoutIndex, requireAvailable: false, null);
            }
            if (!IsShowingOptions) view.SetVisibleLength(line.VisibleLength);
        }

        public void Click()
        {
```

```csharp
using System.Collections.Generic;
using Yarn.Unity;

namespace Game.Core
{
    /// <summary>
    /// Yarn 변수 저장소를 이력 위에 구현한다(설계 87 §4.2). 참/거짓 변수(`$helped_cart`)는 플래그(`helped_cart`)와 같은 값이다 - 조건 열의 flag와
    /// 대본의 변수가 한 원천을 보게 하기 위해서다. 숫자·문자열은 스크립트 변수로만 둔다.
    /// </summary>
    public class YarnEncounterVariableStorage : VariableStorageBehaviour
    {
        private IEncounterHistoryReader reader;
        private IEncounterHistoryWriter writer;
        private IEncounterScriptVariableStore scripts;

        public void Bind(IEncounterHistoryReader reader, IEncounterHistoryWriter writer, IEncounterScriptVariableStore scripts)
        {
            this.reader = reader;
            this.writer = writer;
            this.scripts = scripts;
        }

        private static string FlagOf(string variableName) => variableName.StartsWith("$") ? variableName.Substring(1) : variableName;

        public override bool TryGetValue<T>(string variableName, [System.Diagnostics.CodeAnalysis.NotNullWhen(true)] out T? result)
        {
            if (typeof(T) == typeof(bool) || typeof(T) == typeof(object))
            {
                if (reader != null && reader.HasFlag(FlagOf(variableName)))
                {
                    result = (T)(object)true;
                    return true;
                }
            }
            if (scripts != null && scripts.TryGetScriptValue(variableName, out var value) && value is T typed)
            {
                result = typed;
                return true;
            }
            result = default;
            return false;
        }

        public override void SetValue(string variableName, string stringValue) => scripts?.SetScriptValue(variableName, stringValue);
        public override void SetValue(string variableName, float floatValue) => scripts?.SetScriptValue(variableName, floatValue);
        public override void SetValue(string variableName, bool boolValue) => writer?.SetFlag(FlagOf(variableName), boolValue);

        public override void Clear() => writer?.ResetAll();

        public override bool Contains(string variableName)
            => (reader != null && reader.HasFlag(FlagOf(variableName))) || (scripts != null && scripts.TryGetScriptValue(variableName, out _));

        public override void SetAllVariables(FloatDictionary floats, StringDictionary strings, BoolDictionary bools, bool clear = true)
        {
            if (clear) Clear();
            foreach (var kv in floats) SetValue(kv.Key, kv.Value);
            foreach (var kv in strings) SetValue(kv.Key, kv.Value);
            foreach (var kv in bools) SetValue(kv.Key, kv.Value);
        }

        public override (FloatDictionary FloatVariables, StringDictionary StringVariables, BoolDictionary BoolVariables) GetAllVariables()
        {
            // FloatDictionary 등은 문자열 키 사전이다(정의는 Task 1 Step 4에서 대조 - 생성 방식이 다르면 맞춘다).
            var floats = new FloatDictionary();
            var strings = new StringDictionary();
            var bools = new BoolDictionary();
            if (scripts != null)
            {
                foreach (var kv in scripts.ScriptValues)
                {
                    if (kv.Value is float f) floats[kv.Key] = f;
                    else if (kv.Value is string s) strings[kv.Key] = s;
                }
            }
            if (reader != null) foreach (var flag in reader.ActiveFlags) bools["$" + flag] = true;
            return (floats, strings, bools);
        }
    }
}
```

```csharp
using System;
using UnityEngine;
using Yarn.Unity;

namespace Game.Core
{
    /// <summary>
    /// IDialogueScriptRunner·IDialogueCommandRegistry의 Yarn 구현(설계 87 §7). Bootstrap 상주. 한 번에 대본 하나만 실행한다(대화·이벤트는 동시에 돌지
    /// 않는다 - 이벤트가 뜨면 대화를 먼저 끝낸다). onFinished는 한 번만 부른다.
    /// </summary>
    public class YarnDialogueScriptRunner : MonoBehaviour, IDialogueScriptRunner, IDialogueCommandRegistry, IManagedComponent
    {
        [SerializeField] private DialogueRunner runner;
        [SerializeField] private YarnEncounterVariableStorage storage;
        [SerializeField] private YarnDialoguePresenterBridge bridge;

        private Action<bool> onFinished;
        private bool stopping;

        public bool IsRunning => runner != null && runner.IsDialogueRunning;

        public void RegisterSelf(IDependencyRegistrar registrar)
        {
            registrar.Register<IDialogueScriptRunner>(this);
            registrar.Register<IDialogueCommandRegistry>(this);
        }

        public void ResolveDependencies(IDependencyResolver registrar)
        {
            if (runner == null || storage == null || bridge == null)
            {
                Debug.LogWarning($"{nameof(YarnDialogueScriptRunner)}: Yarn 구성 요소가 연결되지 않았다(Tools > Game > Build Bootstrap Scene).");
                return;
            }
            registrar.TryResolve<IEncounterHistoryReader>(out var reader);
            registrar.TryResolve<IEncounterHistoryWriter>(out var writer);
            registrar.TryResolve<IEncounterScriptVariableStore>(out var scripts);
            storage.Bind(reader, writer, scripts);
            runner.onDialogueComplete.RemoveListener(HandleComplete);
            runner.onDialogueComplete.AddListener(HandleComplete);
        }

        // NodeNames의 구체 타입에 묶이지 않도록 열거로 확인한다.
        public bool HasNode(string node) => runner != null && runner.YarnProject != null && runner.YarnProject.NodeNames != null
            && System.Linq.Enumerable.Contains(runner.YarnProject.NodeNames, node);

        public bool TryRun(string node, IDialoguePresentationTarget target, Action<bool> onFinished)
        {
            if (runner == null || IsRunning || !HasNode(node)) return false;
            bridge.Target = target;
            this.onFinished = onFinished;
            stopping = false;
            // StartDialogue·Stop은 async YarnTask다 - 완료는 onDialogueComplete로 받으므로 기다리지 않는다.
            _ = runner.StartDialogue(node);
            return true;
        }

        public void Stop()
        {
            if (!IsRunning) return;
            stopping = true;
            _ = runner.Stop();
            Finish(false);
        }

        public void AddCommand(string name, Delegate handler) => runner?.AddCommandHandler(name, handler);
        public void AddFunction(string name, Delegate implementation) => runner?.AddFunction(name, implementation);

        private void HandleComplete()
        {
            if (stopping) return;
            Finish(true);
        }

        private void Finish(bool completed)
        {
            var callback = onFinished;
            onFinished = null;
            bridge.Target = null;
            callback?.Invoke(completed);
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    public sealed class YarnScriptLine
    {
        public int LineNumber;
        public string SpeakerId = string.Empty;
        public string Text = string.Empty;
        public List<string> Tags = new();
        public bool IsOption;
        public bool IsCommand;
        public string CommandName = string.Empty;
    }

    public sealed class YarnScriptNode
    {
        public string Title = string.Empty;
        public List<YarnScriptLine> Lines = new();
    }

    /// <summary>
    /// .yarn 텍스트를 노드·줄 구조로 읽는 가벼운 스캐너(설계 87 §9). Yarn 컴파일러에 의존하지 않아 EditMode 콘텐츠 검증(화자·태그·명령 규칙)을
    /// Game.Core 테스트에서 돌릴 수 있다. 문법 전체가 아니라 검증에 필요한 것만 본다: 노드 제목, 화자 접두(`id:`), `#태그`, 선택지(`->`), 명령(`<<x>>`).
    /// 선택지 줄의 `<<if ...>>`는 조건이라 명령으로 세지 않는다.
    /// </summary>
    public static class YarnSourceScanner
    {
        public static List<YarnScriptNode> Scan(string source)
        {
            var nodes = new List<YarnScriptNode>();
            if (string.IsNullOrEmpty(source)) return nodes;

            YarnScriptNode current = null;
            var inBody = false;
            var rawLines = source.Replace("\r\n", "\n").Split('\n');
            for (var i = 0; i < rawLines.Length; i++)
            {
                var line = rawLines[i].Trim();
                if (!inBody)
                {
                    if (line.StartsWith("title:", StringComparison.Ordinal))
                    {
                        current = new YarnScriptNode { Title = line.Substring(6).Trim() };
                    }
                    else if (line == "---" && current != null)
                    {
                        inBody = true;
                    }
                    continue;
                }

                if (line == "===")
                {
                    nodes.Add(current);
                    current = null;
                    inBody = false;
                    continue;
                }
                if (line.Length == 0 || line.StartsWith("//", StringComparison.Ordinal)) continue;

                current.Lines.Add(ParseLine(line, i + 1));
            }
            return nodes;
        }

        private static YarnScriptLine ParseLine(string line, int lineNumber)
        {
            var result = new YarnScriptLine { LineNumber = lineNumber };

            if (line.StartsWith("->", StringComparison.Ordinal))
            {
                result.IsOption = true;
                line = line.Substring(2).Trim();
            }
            else if (line.StartsWith("<<", StringComparison.Ordinal))
            {
                result.IsCommand = true;
                var end = line.IndexOf(">>", StringComparison.Ordinal);
                var body = (end > 2 ? line.Substring(2, end - 2) : line.Substring(2)).Trim();
                var space = body.IndexOf(' ');
                result.CommandName = space > 0 ? body.Substring(0, space) : body;
                return result;
            }

            var hash = line.IndexOf('#');
            var content = hash >= 0 ? line.Substring(0, hash) : line;
            if (hash >= 0)
            {
                foreach (var part in line.Substring(hash).Split('#'))
                {
                    var tag = part.Trim();
                    if (tag.Length > 0) result.Tags.Add(tag);
                }
            }

            // 선택지의 조건(`<<if ...>>`)은 본문에서 뗀다.
            var condition = content.IndexOf("<<", StringComparison.Ordinal);
            if (condition >= 0) content = content.Substring(0, condition);
            content = content.Trim();

            if (!result.IsOption)
            {
                var colon = content.IndexOf(':');
                if (colon > 0 && content.Substring(0, colon).IndexOf(' ') < 0)
                {
                    result.SpeakerId = content.Substring(0, colon).Trim();
                    content = content.Substring(colon + 1).Trim();
                }
            }
            result.Text = content;
            return result;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace Game.Core
{
    /// <summary>
    /// 콘텐츠 정합성 검사(설계 87 §9). 테이블: YarnNode 존재. 대본: 화자 Id 존재, 선택지 묶음마다 `#timeout` 0~1개, 묶음 바로 앞 줄에 `#ask`(대화) 또는
    /// `#choice`(이벤트) - Yarn은 줄이 끝나야 선택지를 넘겨주므로 질문 줄이 대기 없이 끝나야 선택지가 곧바로 뜬다. 좌우 자리 규칙(88)·명령 규칙(89)은 각
    /// 설계 제작 때 여기에 더한다.
    /// </summary>
    public static class EncounterContentValidator
    {
        public static List<string> ValidateTable(IReadOnlyList<EventEntry> events, IReadOnlyList<DialogueEntry> dialogues, Func<string, bool> nodeExists)
        {
            var errors = new List<string>();
            foreach (var e in events)
            {
                if (!nodeExists(e.YarnNode)) errors.Add($"이벤트 '{e.Id}': Yarn 노드 '{e.YarnNode}'가 없다.");
            }
            foreach (var d in dialogues)
            {
                if (!nodeExists(d.YarnNode)) errors.Add($"대화 '{d.Id}': Yarn 노드 '{d.YarnNode}'가 없다.");
            }
            return errors;
        }

        public static List<string> ValidateScripts(IReadOnlyList<YarnScriptNode> nodes, Func<string, bool> speakerExists)
        {
            var errors = new List<string>();
            foreach (var node in nodes)
            {
                var lines = node.Lines;
                for (var i = 0; i < lines.Count; i++)
                {
                    var line = lines[i];
                    if (!line.IsOption && !line.IsCommand && line.SpeakerId.Length > 0 && !speakerExists(line.SpeakerId))
                    {
                        errors.Add($"{node.Title}:{line.LineNumber} 화자 '{line.SpeakerId}'가 캐릭터·외부 인물에 없다.");
                    }

                    var startsGroup = line.IsOption && (i == 0 || !lines[i - 1].IsOption) && !IsInsideOptionBody(lines, i);
                    if (!startsGroup) continue;

                    var prev = PreviousNonOptionLine(lines, i);
                    if (prev == null || !(prev.Tags.Contains(DialogueTags.Ask) || prev.Tags.Contains(DialogueTags.Choice)))
                    {
                        errors.Add($"{node.Title}:{line.LineNumber} 선택지 앞 줄에 #ask(대화) 또는 #choice(이벤트)가 없다.");
                    }

                    var timeouts = 0;
                    foreach (var option in OptionGroup(lines, i)) if (option.Tags.Contains(DialogueTags.Timeout)) timeouts++;
                    if (timeouts > 1) errors.Add($"{node.Title}:{line.LineNumber} 선택지 묶음에 #timeout이 {timeouts}개 - 0~1개여야 한다.");
                }
            }
            return errors;
        }

        // 스캐너는 들여쓰기를 버리므로 "선택지 → 본문 줄들 → 다음 선택지"를 한 묶음으로 본다: 앞쪽으로 같은 묶음의 선택지가 있으면 묶음 시작이 아니다.
        private static bool IsInsideOptionBody(List<YarnScriptLine> lines, int index)
        {
            for (var j = index - 1; j >= 0; j--)
            {
                if (lines[j].IsOption) return true;
                if (lines[j].Tags.Contains(DialogueTags.Ask) || lines[j].Tags.Contains(DialogueTags.Choice)) return false;
            }
            return false;
        }

        private static YarnScriptLine PreviousNonOptionLine(List<YarnScriptLine> lines, int index)
        {
            for (var j = index - 1; j >= 0; j--)
            {
                if (!lines[j].IsOption && !lines[j].IsCommand) return lines[j];
            }
            return null;
        }

        private static IEnumerable<YarnScriptLine> OptionGroup(List<YarnScriptLine> lines, int start)
        {
            for (var j = start; j < lines.Count; j++)
            {
                if (lines[j].Tags.Contains(DialogueTags.Ask) || lines[j].Tags.Contains(DialogueTags.Choice)) yield break;
                if (lines[j].IsOption) yield return lines[j];
            }
        }
    }
}
```
